add_namespace = portugal

country_event = {# Britain asks Portugal for military access
	id = portugal.1
	title = portugal.1.t
	desc = portugal.1.d
	picture = GFX_report_event_diplomatic_message

	is_triggered_only = yes

	fire_only_once = yes

	option = { # Okay mister
		name = portugal.1.a
		ai_chance = {
			base = 100
		}
		ENG = { 
			country_event = portugal.2
		}		
		every_country = {
			limit = { 
				OR = { 
					is_in_faction_with = ENG 
					tag = ENG
				}
			}
			if = {
				limit = {
					NOT = { has_military_access_to = ROOT }
				}
				ROOT = { give_military_access = PREV }
			}
		}
	} 
	option = { # No Mister
		name = portugal.1.b
		ai_chance = {
			base = 0
		}
	} 
}

	} 
}
