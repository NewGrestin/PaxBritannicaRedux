			is_subject = no
		}

			base = 100
			modifier = {
				factor = 0
			}
		}

		if = {
			limit = {
				ROOT = {
					has_monarchist_government = yes

			limit = {

					NOT = {
						original_tag = ROC
					}
				}
			}
			MON = {
				drop_cosmetic_tag = yes
				set_cosmetic_tag = MON_Occupied_Republic
			}
		}

			limit = {

					has_republican_government = yes
					original_tag = ROC
				}
			}
			MON = {
				drop_cosmetic_tag = yes
				set_cosmetic_tag = MON_Occupied_Republic_ROC

		recheck_annexations = yes

		log = "[GetLogInfo]: event annexation.7000 option annexation.give_to_republic_of_china"

		trigger = {
			ROC = {
				exists = yes
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				annexations_can_annex = yes
			}
		}
		ROC = {
			annex_country = {
				target = MON
				transfer_troops = no
			}
		}
		recheck_annexations = yes
	}
	
	option = { #give to Taiping
		log = "[GetLogInfo]: event annexation.7000 option annexation.give_to_china"
