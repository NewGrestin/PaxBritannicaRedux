country_event = { #Fate of New Georgia
	id = annex.5001
	title = annexation_285
	desc = annexation_285_desc
	picture = GFX_report_PB_Peace_Deals_Europe

	is_triggered_only = yes

	immediate = { 
		hidden_effect = { 
			if = { 
				limit = { 
					OR = { 
						is_imperial_federation = yes
						is_australian_tag = yes
					}
				}
				save_event_target_as = can_annex
			}
			every_state = {
				remove_claim_by = CNG
				if = {
					limit = { is_new_georgia = yes }
					add_core_of = CNG
				}
				else = {
					remove_core_of = CNG
				}
			}
			CNG = { 
				remove_ideas = { 
					PB_CNG_Shades_Of_Red
				}
			}
		}
	}
	
	option = { #Release New Georgia Whole
		log = "[GetLogInfo]: event annex.5001 option annex.5001.a"
		name = annex.5001.a
		trigger = {
			is_subject = no
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				annexations_can_annex = yes
			}
		}

		if = {
			limit = { ROOT = { has_monarchist_government = yes } }
			CNG = {
				set_cosmetic_tag = CNG_Occupied_Monarchy
			}
		}
		
		if = {
			limit = { ROOT = { has_republican_government = yes } }
			CNG = {
				set_cosmetic_tag = CNG_Occupied_Republic
			}
		}
		set_temp_variable = { tag_to_release = CNG }
		every_state = {
			limit = { is_core_of = CNG }
			remove_claim_and_add_core_of_target = yes
		}
		release_targeted_tag = yes		
	}
	
	option = { #Give to overlord
		log = "[GetLogInfo]: event annex.5001 option annex.give_to_overlord"
		name = annex.give_to_overlord
		trigger = {
			annexations_should_give_lands_to_overlord = yes
		}
		set_temp_variable = { tag_to_release = overlord }		
		every_state = {
			limit = { is_new_georgia = yes }
			transfer_if_owned = yes
		}
	}

	option = { #Integrate
		log = "[GetLogInfo]: event annex.5001 option annex.integration"
		name = annex.integration
		trigger = {
			annexations_can_annex = yes
		}
		every_owned_state = { 
			limit = { 
				is_new_georgia = yes
				is_valid_annexation_state = yes
			}
			add_claim_of_ROOT = yes
		}
	}

	option = { #Cancel
		log = "[GetLogInfo]: event annex.5001 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
		recheck_annexations = yes
    }
	
	after = {
		recheck_annexations = yes
	}
}
