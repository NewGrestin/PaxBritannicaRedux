country_event = { # Fate of Acadia
	id = annex.2000
	title = annexation_371
	desc = annexation_371_desc
	picture = GFX_report_event_Generic_Art_Gallery

	is_triggered_only = yes

	immediate = {
		# Save UC or FSA as grant targets if ROOT can grant land
		if = {
			limit = { AME = { ROOT_can_grant_land = yes } }
			AME = { save_event_target_as = america }
		}
	}

	option = { # Release Acadia
		log = "[GetLogInfo]: event annex.2000 option annex.2000.a"
		name = annex.2000.a
		trigger = { is_subject = no }
		ai_chance = {
			base = 65
			modifier = { factor = 0 annexations_can_annex = yes }
		}

		set_temp_variable = { tag_to_release = ACD } # Acadia tag
		every_state = {
			limit = { is_acadia = yes }
			remove_claim_and_add_core_of_target = yes
		}
		release_targeted_tag = yes

		if = { limit = { ROOT = { has_monarchist_government = yes } } ACD = { set_cosmetic_tag = ACD_Occupied_Monarchy } }
		if = { limit = { ROOT = { has_republican_government = yes } } ACD = { set_cosmetic_tag = ACD_Occupied_Republic } }
	}

	option = { # Give to America
		log = "[GetLogInfo]: event annex.2000 option annex.give_to_america"
		name = annex.america.give_to_america
		trigger = { has_event_target = america }
		ai_chance = {
			base = 100
			modifier = { factor = 0 annexations_can_annex = yes }
			modifier = { factor = 0.1 event_target:america = { is_subject = yes } }
		}

		set_temp_variable = { tag_to_release = event_target:america }
		every_state = {
			limit = { is_acadia = yes }
			transfer_if_owned = yes
		}
	}

	option = { # Give to Overlord
		log = "[GetLogInfo]: event annex.2000 option annex.give_to_overlord"
		name = annex.give_to_overlord
		trigger = { annexations_should_give_lands_to_overlord = yes }
		set_temp_variable = { tag_to_release = overlord }
		every_state = {
			limit = { is_acadia = yes }
			transfer_if_owned = yes
		}
	}

	option = { # Integrate
		log = "[GetLogInfo]: event annex.2000 option annex.integration"
		name = annex.integration
		trigger = { annexations_can_annex = yes }
		every_state = {
			limit = { is_acadia = yes }
			transfer_if_owned = yes
		}
	}

	option = { # Cancel
		log = "[GetLogInfo]: event annex.2000 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
	}

	after = { recheck_annexations = yes }
}

	title = annexation_apacheria
	desc = annexation_apacheria_desc
