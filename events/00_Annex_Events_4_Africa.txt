country_event = { #Fate of Morocco
	id = annex.4000
	title = annexation_morocco
	desc = annexation_morocco_desc
	picture = GFX_report_PB_Peace_Deals_Europe

	is_triggered_only = yes

	immediate = {
		807 = { retain_state = yes } #Reguibat
	}
	
	option = { #Release Morocco Whole
		log = "[GetLogInfo]: event annex.4000 option annex.4000.a"
		name = annex.4000.a
		trigger = {
			is_subject = no
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				annexations_can_annex = yes
			}
		}

		set_temp_variable = { tag_to_release = MOR }
		every_state = {
			limit = { is_core_of = MOR }
			remove_claim_and_add_core_of_target = yes
		}
		release_targeted_tag = yes

		if = {
			limit = {
				ROOT = {
					has_monarchist_government = yes
				}
			}
			MOR = {
				drop_cosmetic_tag = yes
				set_cosmetic_tag = MOR_Occupied_Monarchy
			}
		}

		if = {
			limit = {
				ROOT = {
					has_republican_government = yes
				}
			}
			MOR = {
				drop_cosmetic_tag = yes
				set_cosmetic_tag = MOR_Occupied_Republic
			}
		}

		if = {
			limit = {
				ROOT = {
					tag = SPR
				}
			}
			MOR = {
				drop_cosmetic_tag = yes
				set_cosmetic_tag = MOR_SPR_Custody
			}
		}
	}

	option = { #Cancel
		log = "[GetLogInfo]: event annex.4000 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
		recheck_annexations = yes
    }
}

country_event = { #Fate of Reguibat
    id = annex.4001
    title = annexation_reguibat
    desc = annexation_reguibat_desc
    picture = GFX_report_PB_Peace_Deals_Europe

    is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				OR = {
					461 = { is_claim_or_core_of_ROOT = yes } #Casablanca
					920 = { is_claim_or_core_of_ROOT = yes } #El eglab
				}
			}
			save_event_target_as = can_annex
		}
		if = {
			limit = {
				461 = { #Casablanca
					owner = { ROOT_can_grant_land = yes }
				}
			}
			var:461.owner = { save_event_target_as = morocco }
		}
		if = {
			limit = {
				920 = { #El eglab
					annexations_state_owner_is_valid_grant_target = yes
					owner = {
						NOT = { tag = event_target:morocco }
						ROOT_can_grant_land = yes
					}
				}
			}
			var:920.owner = { save_event_target_as = carnegia }
		}
	}

    option = { #Give it to Morocco
        log = "[GetLogInfo]: event annex.4001 option annex.4001.a"
        name = annex.4001.a
        trigger = {
			has_event_target = morocco
		}
		ai_chance = {
			base = 40
			modifier = {
				factor = 0
				annexations_AI_wants_to_annex = yes
			}
			modifier = {
				factor = 0
				NOT = { is_ally_with = event_target:morocco }
				is_ally_with = event_target:carnegia
			}
			modifier = {
				factor = 2
				NOT = { has_event_target = carnegia }
			}
			modifier = {
				factor = 0.1
				event_target:morocco = { is_subject = yes }
			}
		}
		set_temp_variable = { tag_to_release = event_target:morocco }
		807 = { transfer_state_and_notify = yes } #Reguibat
    }

    option = { #Give it to Carnegia
        log = "[GetLogInfo]: event annex.4001 option annex.4001.b"
        name = annex.4001.b
        trigger = {
			has_event_target = carnegia
		}
		ai_chance = {
			base = 40
			modifier = {
				factor = 0
				annexations_AI_wants_to_annex = yes
			}
			modifier = {
				factor = 0
				NOT = { is_ally_with = event_target:carnegia }
				is_ally_with = event_target:morocco
			}
			modifier = {
				factor = 2
				NOT = { has_event_target = morocco }
			}
			modifier = {
				factor = 0.1
				event_target:carnegia = { is_subject = yes }
			}
		}
		set_temp_variable = { tag_to_release = event_target:carnegia }
		807 = { transfer_state_and_notify = yes } #Reguibat
    }

	option = { #Give to overlord
		log = "[GetLogInfo]: event annex.4001 option name annex.give_to_overlord"
		name = annex.give_to_overlord
		trigger = {
			annexations_should_give_lands_to_overlord = yes
		}
		set_temp_variable = { tag_to_release = overlord }
		807 = { transfer_state_and_notify = yes } #Reguibat
	}

	option = { #Annex
		log = "[GetLogInfo]: event annex.4001 option name annex.integration"
		name = annex.integration
		trigger = {
			annexations_can_annex = yes
		}
		807 = { transfer_state_and_notify = yes } #Reguibat
	}

	option = { #Cancel
		log = "[GetLogInfo]: event annex.4001 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
    }

	after = {
		recheck_annexations = yes
	}
}
