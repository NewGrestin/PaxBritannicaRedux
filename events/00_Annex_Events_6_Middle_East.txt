	id = annex.6000
	title = annexation_417
	desc = annexation_417_desc
	picture = GFX_report_PB_Peace_Deals_Asia

	is_triggered_only = yes
	
	immediate = {
		hidden_effect = { 
			PER = { 
				remove_ideas = { 
					PB_PER_Ummah_Vanguard 
					PB_PER_Uneasy_Sisterhood
				}
			}
			every_state = {
				remove_claim_by = PER
				if = {
					limit = { is_persia = yes }
					add_core_of = PER
				}
				else = {
					remove_core_of = PER
				}
			}
		}
		266 = { retain_state = yes } #Mazandaran
		410 = { retain_state = yes } #Sistan
		413 = { retain_state = yes } #Kuzestan

		420 = { retain_state = yes } #Gilan
		421 = { retain_state = yes } #Kordestan
		1423 = { retain_state = yes } #Lorestan
		1426 = { retain_state = yes } #Orumiyeh
		416 = { retain_state = yes } #Orumiyeh
	}

	option = { #Release Persia
		log = "[GetLogInfo]: event annex.6000 option annex.6000.a"
		name = annex.6000.a
		trigger = {
			is_subject = no
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				annexations_can_annex = yes
			}
		}

		set_temp_variable = { tag_to_release = PER }
		release_targeted_tag = yes

		if = { 
			limit = { has_custom_puppet_PER = yes }
			setup_custom_puppet_PER = yes
			custom_effect_tooltip = has_custom_puppet
		}

		else_if = {
			limit = {
				ROOT = {
					has_monarchist_government = yes
				}
			}
			PER = {
				set_cosmetic_tag = PER_Occupied_Monarchy
				
			}
		}

		else_if = {
			limit = {
				ROOT = {
					has_republican_government = yes
				}
			}
			PER = {
				set_cosmetic_tag = PER_Occupied_Republic	
			}
		}
	}
	
	option = { #Give to overlord
		log = "[GetLogInfo]: event annex.6000 option annex.give_to_overlord"
		name = annex.give_to_overlord
		trigger = {
			annexations_should_give_lands_to_overlord = yes
		}
		set_temp_variable = { tag_to_release = overlord }
		every_owned_state = {
			limit = {
				is_persia = yes
				is_allowed_annexation_state = yes
			}
			transfer_state_and_notify = yes
		}
		PER = { set_transfer_target_cores = yes }
		transfer_targeted_territory = yes
	}

	option = { #Integrate
		log = "[GetLogInfo]: event annex.6000 option annex.integration"
		name = annex.integration
		trigger = {
			annexations_can_annex = yes
		}

			limit = {
				is_persia = yes
			}
			add_claim_of_ROOT = yes
		}
	}

	option = { #Cancel
		log = "[GetLogInfo]: event annex.6000 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
	}
