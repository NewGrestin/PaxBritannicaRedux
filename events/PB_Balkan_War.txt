country_event = { # Balkan War Trigger - HUN/BUL vs the coalition
	id = balkwar.1
	title = balkwar.1.t
	desc = balkwar.1.d
	picture = GFX_report_AC_Edison_Mining

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		GRE = {
			create_faction = anti_hungarian_pact_faction
			add_to_faction = SER
			country_event = { id = balkwar.2 days = 25 }
			country_event = { id = balkwar.6 days = 65 }
		}

		if = { # Hungary unified: HWA doesn't exist
			limit = { NOT = { country_exists = HWA } }
			HUN = {
				create_faction = budapest_entente_faction
				add_to_faction = BUL
				add_to_faction = CRO
			}
		}
		else_if = { # HWA won the civil war: Hungary doesn't exist
			limit = { NOT = { country_exists = HUN } }
			HWA = {
				create_faction = budapest_entente_faction
				add_to_faction = BUL
				add_to_faction = SLA
			}
		}
		else = { # Both exist
			HUN = {
				create_faction = budapest_entente_faction
				add_to_faction = BUL
				add_to_faction = CRO
			}
			if = { # Hungarian civil war still ongoing: HWA joins the Pact
				limit = { HUN = { has_war_with = HWA } }
				GRE = { add_to_faction = HWA }
				set_global_flag = GBW_HWA_Joined_Pacte
			}
		}

		MOL = {
			country_event = { id = balkwar.5 days = 39 }
		}
	}

	option = { # BALKWAR
		name = balkwar.1.a
	}
}
