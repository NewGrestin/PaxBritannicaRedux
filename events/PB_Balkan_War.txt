country_event = { #Balkan War - Bulgarian Defeat
	id = balkwar.3
	title = balkwar.3.t
	desc = balkwar.3.d
	picture = GFX_report_AC_Edison_Mining
	is_triggered_only = yes
	
	trigger = {
		tag = BUL
		has_war_with = GRE
		OR = {
			has_capitulated = yes
		}
	}

	immediate = { 
		BUL = {
			set_country_flag = Balkan_War_Bulgaria_Defeated
			leave_faction = yes
			white_peace = ROM
			white_peace = MOL
			white_peace = GRE
			white_peace = SLA
			white_peace = CRO
			white_peace = ALB
			white_peace = SER
			every_owned_state = {
				limit = { 
					OR = { 
						is_claimed_by = SER 
						is_core_of = SER 
					}
				} 
				transfer_state_to = SER
			}
			every_owned_state = {
				limit = {
					OR = {
						state = 1037 #Ploiesti
						state = 46 #Muntenia
						state = 81 #Oltenia
						state = 1039 #Tulcea
						state = 1038 #Constanta
						state = 77 #Dobruja
						is_claimed_by = MOL
						is_core_of = MOL
					}
				}
				if = {
					limit = { NOT = { is_core_of = MOL } }
					add_claim_by = MOL
				}
				transfer_state_to = MOL
			}
		}
		if = { # If Romania exists
			limit = { 
				ROM = { is_subject_of = BUL }
			}
			ROM = {
				all_core_state  = { 
					add_core_of = MOL
				}
			}
			MOL = { 
				annex_country = { 
					target = ROM
   					transfer_troops = no
				}
			}
		}
	}
	option = { #blah
		name = balkwar.3.a
	}
}
