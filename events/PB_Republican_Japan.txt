		}

			declare_war_on = {
				target = RUS
				type = annex_everything
			}

			declare_war_on = {
				target = MON
				type = annex_everything
			}

			declare_war_on = {
				target = MAN
				type = annex_everything
			}
		}
		if = {
			limit = {
				RUS = {
					is_in_faction_with = FRA
				}
			}
			FRA = {
				add_to_faction = MAN
				add_to_faction = MON
			}
		}
		else = {
			RUS = {
				add_to_faction = MAN
				add_to_faction = MON
			}
		}
	}
}
