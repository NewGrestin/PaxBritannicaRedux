country_event = { #Launch Success
	id = OW_success_events.1
	title = OW_success_events.1.t
	desc = OW_success_events.1.d
	picture = GFX_report_OBW_Success1

	option = { 
		name = OW_success_events.1.a
		
		if = {
		    limit = { check_variable = { rocket_special_module_selected = 0 } } #research module installed
		    custom_override_tooltip = {
				tooltip = {
					localization_key = tooltip_add_orbit_research
					NUM = 15
				}
				add_to_variable = { OW_research_current = 15 }
			}

			limit = { check_variable = { rocket_special_module_selected = 1 } } #communications booster
