country_event = { #Choice of topic
	id = Prague_Conference.5
	title = Prague_Conference.5.t
	desc = Prague_Conference.5.d
	picture = GFX_report_PB_Conference

	is_triggered_only = yes
	
	immediate = {
		
	}

	option = { #Tackle the matter of Central Europe
		name = Prague_Conference.5.a
		trigger = {
			NOT = { has_global_flag = PragueConf_Central_Europe }
		}
		ai_chance = {
			base = 50
		}
		
		add_to_variable = { Central_Europe_prague_conf = 1 }
		
		if = {
			limit = {
				AND = {
					FRA = {
						has_variable = { FRA.prague_conf_choice = 1 }
					}
					RUS = {
						has_variable = { RUS.prague_conf_choice = 1 }
					}
				}
			}
			if = {
				limit = {
					Central_Europe_prague_conf > 1
				}
				ROOT = {
					country_event = {
						id = Prague_Conference.6
						days = 1
					}
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					FRA = {
						has_variable = { FRA.prague_conf_choice = 1 }
					}
					NOT = {
						RUS = {
							has_variable = { RUS.prague_conf_choice = 1 }
						}
					}
				}
				RUS = {
					country_event = {
						id = Prague_Conference.5
						days = 1
					}
				}
			}
		}
		else_if = {
			limit = {
				AND = {
					RUS = {
						has_variable = { FRA.prague_conf_choice = 1 }
					}
					NOT = {
						FRA = {
							has_variable = { RUS.prague_conf_choice = 1 }
						}
					}
				}
				FRA = {
					country_event = {
						id = Prague_Conference.5
						days = 1
					}
				}
			}
		}
	}
	
	option = { #Tackle the matter of the Balkans
		name = Prague_Conference.5.b
		trigger = {
			NOT = { has_global_flag = PragueConf_Balkans }
		}
		ai_chance = {
			base = 50
		}
		
		add_to_variable = { ROOT.Balkans_prague_conf = 1 }
	}
	
	option = { #Tackle the matter of Scandinavia
		name = Prague_Conference.5.c
		trigger = {
			NOT = { has_global_flag = PragueConf_Scandinavia }
		}
		ai_chance = {
			base = 50
		}
		
		add_to_variable = { ROOT.Scandinavia_prague_conf = 1 }
	}
	
	option = { #Tackle the matter of the Levant
		name = Prague_Conference.5.e
		trigger = {
			NOT = { has_global_flag = PragueConf_Levant }
		}
		ai_chance = {
			base = 50
		}
		
		add_to_variable = { ROOT.Levant_prague_conf = 1 }
	}
	
	option = { #Let the other chose to preserve our leverages
		name = Prague_Conference.5.f
		trigger = {
			NOT = { has_global_flag = PragueConf_Levant }
		}
		ai_chance = {
			base = 50
		}
		
		add_to_variable = { ROOT.diplo_weight = 5 }
	}
	
	after = {
		add_to_variable = { ROOT.prague_conf_choice = 1 }
	}
