country_event = { #Fate of Venezuela
	id = annex.3000
	title = annexation_venezuela
	desc = annexation_venezuela_desc
	picture = GFX_report_PB_Peace_Deals_Europe

	is_triggered_only = yes
	
	option = { #Release Venezuela Whole
		log = "[GetLogInfo]: event annex.3000 option annex.3000.a"
		name = annex.3000.a
		trigger = {
			is_subject = no
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				annexations_can_annex = yes
			}
		}

		set_temp_variable = { tag_to_release = VEN }
		every_state = {
			limit = { is_core_of = VEN }
			remove_claim_and_add_core_of_target = yes
		}
		release_targeted_tag = yes

		if = {
			limit = { ROOT = { has_monarchist_government = yes } }
			VEN = {
				set_cosmetic_tag = VEN_Occupied_Monarchy
			}
		}

		if = {
			limit = { ROOT = { has_republican_government = yes } }
			VEN = {
				set_cosmetic_tag = VEN_Occupied_Republic
			}
		}
	}

	option = { #Give to overlord
		log = "[GetLogInfo]: event annex.3000 option annex.give_to_overlord"
		name = annex.give_to_overlord
		trigger = {
			annexations_should_give_lands_to_overlord = yes
		}
		set_temp_variable = { tag_to_release = overlord }
		every_owned_state = {
			limit = {
				is_allowed_annexation_state = yes
			}
			transfer_state_and_notify = yes
		}
		recheck_annexations = yes
	}

	option = { #Annex
		log = "[GetLogInfo]: event annex.3000 option name annex.integration"
		name = annex.integration
		trigger = {
			annexations_can_annex = yes
		}
		every_owned_state = {
			limit = { is_core_of = VEN }
			add_claim_of_ROOT = yes
		}
	}

	option = { #Cancel
		log = "[GetLogInfo]: event annex.3000 option annex.cancel"
		name = annex.cancel
		ai_chance = { base = 0 }
		
    }

	after = {
		recheck_annexations = yes
	}
}
