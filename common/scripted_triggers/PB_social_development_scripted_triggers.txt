check_socdev_campaign_cap = {
	check_variable = { socdev_active_campaigns < socdev_active_campaigns_cap }
}
socdev_POV_campaign_1_check = {
    NOT = {
        OR = {
            has_idea = SD_POV_campaign_2_dummy
            has_country_flag = SD_POV_campaign_1_cooldown
        }
    }
    check_expr = {
        value = var:political_power
        greater_than_or_equals = THIS.SD_campaigns_pp_cost
    }
    check_socdev_campaign_cap = yes
}
