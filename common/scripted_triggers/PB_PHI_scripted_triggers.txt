PHI_is_phase_1_trigger = { 
	custom_trigger_tooltip = {
		tooltip = GXC_is_phase_1_tt
		PHI = {
			has_country_flag = PHI_is_phase_1_trigger
			NOT = {
				has_country_flag = PHI_is_phase_2_trigger
				has_country_flag = PHI_is_phase_3_trigger
			}
		}
	}
}
