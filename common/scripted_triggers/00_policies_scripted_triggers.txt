### triggers below allow policy to be changed manually only by one level up or down
### those are going in order left to right and top to down rows of laws
### from the "lowest ranked" to the "top ranked"

can_select_no_voting_policy = {
	custom_override_tooltip = {
		tooltip = can_change_policy_by_one_level_tt
		has_idea = PB_Single_Party_State
		OR = {
			NOT = { has_idea = PB_Managed_Democracy }
			NOT = { has_idea = PB_Aristocratic_Democracy }
			NOT = { has_idea = PB_Limited_Disenfranchisement }
			NOT = { has_idea = PB_Universal_Citizenship }
		}
	}
	custom_override_tooltip = {
		tooltip = have_changed_civilian_policy_last_180_days_tt
		NOT = { has_country_flag = have_changed_civilian_policy_last_180_days }
	}
}
