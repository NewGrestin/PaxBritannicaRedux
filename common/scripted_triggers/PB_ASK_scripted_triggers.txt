ASK_check_CC_assimilate_available = {
	custom_trigger_tooltip = {
		tooltip = ASK_Available_CC_To_Assimilate	
		OR = {
			AND = {
				check_variable = { ASK_Russian_CC_Status > 1 }
				check_variable = { ASK_Russian_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Polish_CC_Status > 1 }
				check_variable = { ASK_Polish_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Ukrainian_CC_Status > 1 }
				check_variable = { ASK_Ukrainian_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Kazakh_CC_Status > 1 }
				check_variable = { ASK_Kazakh_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Jewish_CC_Status > 1 }
				check_variable = { ASK_Jewish_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Native_CC_Status > 1 }
				check_variable = { ASK_Native_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_Manchu_CC_Status > 1 }
				check_variable = { ASK_Manchu_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_American_CC_Status > 1 }
				check_variable = { ASK_American_CC_Status < 4 }
			}
			AND = {
				check_variable = { ASK_British_CC_Status > 1 }
				check_variable = { ASK_British_CC_Status < 4 }
			}
		}
	}
}
