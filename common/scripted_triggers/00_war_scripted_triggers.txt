at_peace_or_same_enemies_as_TARGET = {
	OR = {
		AND = {
			has_war = no
			NOT = { any_claim = yes }
		}
		all_enemy_country = { has_war_with = var:target }
	}
	hidden_trigger = { NOT = { has_country_flag = imminent_war } }
}

at_peace_or_same_enemies_as_FROM = {
	set_temp_variable = { target = FROM }
	at_peace_or_same_enemies_as_TARGET = yes
}

at_peace_or_same_enemies_as_ROOT = {
	set_temp_variable = { target = ROOT }
	at_peace_or_same_enemies_as_TARGET = yes
}

at_peace_or_same_major_enemies_as_TARGET = {
	OR = {
		AND = {
			has_war = no
			NOT = { any_claim = yes }
		}
		all_enemy_country = {

					is_in_faction = no
				}
			}
		}
	}
	#hidden_trigger = { NOT = { has_country_flag = imminent_war } }
}

can_unlock_new_special_forces_tree = {
	if = {
		limit = { NOT = { has_variable = sf_trees } }
		custom_trigger_tooltip = {
			tooltip = can_unlock_new_special_forces_tree_none
			NOT = { has_variable = sf_trees }
		}
	}
	else_if = {
		limit = { check_variable = { sf_trees > 0 } }
		custom_trigger_tooltip = {
			tooltip = can_unlock_new_special_forces_tree_more_yes
			check_variable = { sf_trees > 0 }
		}
	}
	else = {
		custom_trigger_tooltip = {
			tooltip = can_unlock_new_special_forces_tree_more_no
			always = no
		}
	}
}
