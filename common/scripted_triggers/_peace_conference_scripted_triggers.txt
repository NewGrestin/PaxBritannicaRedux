is_ambition_area_of_ROOT = {
	OR = {
		has_variable = ambition_area_@ROOT
		AND = {
			ROOT = {
				is_actual_major = yes
				OR = {
					tag = GER
					tag = AME
					tag = RUS
					tag = ENG
					tag = FRA
					tag = JAP
				}
			}
			is_important_state = yes
		}
	}
}
