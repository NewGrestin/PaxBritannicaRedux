### BUILD FACILITIES LOGIC ###
default_build_nuclear_facility = {
	enable = {
		has_tech = atomic_research
		num_of_civilian_factories > 30
		num_of_military_factories > 30
		nuclear_facility < 1
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = nuclear_facility
		value = 100
	}
}

default_build_land_facility = {
	enable = {
		is_major = yes
		num_of_civilian_factories > 40
		num_of_military_factories > 40
		OR = {
			has_tech = basic_light_tank_chassis
			has_tech = basic_medium_tank_chassis
			has_tech = basic_heavy_tank_chassis
			has_tech = basic_duopod_automat_chassis
			has_tech = basic_tripod_automat_chassis
			has_tech = basic_quadrupod_automat_chassis
		}
		land_facility < 1
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = land_facility
		value = 1
	}
}

### RECRUIT SCIENTISTS LOGIC ###
default_recruit_land_scientist = {
	allowed = {
		has_dlc = "No Compromise, No Surrender"
	}
	enable = {
		OR = {
			has_facility_specialization = specialization_land
			can_assign_supportive_scientist_to_faction = specialization_land
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = recruit_scientist
		id = specialization_land
		value = 100
	}
}

default_recruit_land_scientist_no_ncns = {
	allowed = {
		NOT = { has_dlc = "No Compromise, No Surrender" }
	}
	enable = {
		has_facility_specialization = specialization_land
		NOT = { has_scientist_specialization = specialization_land }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = recruit_scientist
		id = specialization_land
		value = 100
	}
}
