### BUILD FACILITIES LOGIC ###
default_build_nuclear_facility = {
	enable = {
		has_tech = atomic_research
		NOT = { has_facility_specialization = specialization_nuclear }
		num_of_civilian_factories > 40
		num_of_military_factories > 40
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = nuclear_facility
		value = 10
	}
}


default_build_naval_facility = {
	enable = {
		is_major = yes
		NOT = { has_facility_specialization = specialization_naval }
		num_of_naval_factories > 25
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = naval_facility
		value = 10
	}
}

default_build_land_facility = {
	enable = {
		is_major = yes
		NOT = { has_facility_specialization = specialization_land }
		num_of_civilian_factories > 35
		num_of_military_factories > 35
		OR = {
			has_tech = improved_light_tank_chassis
			has_tech = improved_medium_tank_chassis
			has_tech = improved_heavy_tank_chassis
			has_tech = improved_duopod_automat_chassis
			has_tech = improved_tripod_automat_chassis
			has_tech = improved_quadrupod_automat_chassis
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = land_facility
		value = 10
	}
}

default_build_air_facility = {
	enable = {
		is_major = yes
		NOT = { has_facility_specialization = specialization_air }
		num_of_military_factories > 35
		
		OR = {
			has_tech = improved_small_airframe
			has_tech = improved_medium_airframe
			has_tech = improved_large_airframe
			has_tech = improved_transport_helicopter
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = build_building
		id = air_facility
		value = 10
	}
}

### RECRUIT SCIENTISTS LOGIC ###
default_recruit_land_scientist = {
	allowed = {
		has_dlc = "No Compromise, No Surrender"
	}
	enable = {
		OR = {
			has_facility_specialization = specialization_land
			can_assign_supportive_scientist_to_faction = specialization_land
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = recruit_scientist
		id = specialization_land
		value = 100
	}
}
