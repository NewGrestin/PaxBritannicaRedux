ignore_andorra = {
	allowed = {
		original_tag = ROOT
	}
	enable = {
		country_exists = ADR
		ROOT = {
			NOT = {
				has_war_with = ADR
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = ignore
		id = "ADR"
		value = 2000
	}

	ai_strategy = {
		type = front_unit_request
		tag = ADR
		value = -4000
	}
}
