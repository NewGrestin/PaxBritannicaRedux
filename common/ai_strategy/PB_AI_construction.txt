
default_build_tesla_towers = {
	enable = {
		has_war_with_major = no
		is_major = yes
		num_of_civilian_factories > 70
		num_of_military_factories > 100
		tesla_tower < 16
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = building_target
		id = tesla_tower
		value = 15
	}
}
