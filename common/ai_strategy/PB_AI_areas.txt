### GENERAL STRATEGIES ###
less_focus_on_non_neighbour_fronts = {
	enable = {
		is_in_faction = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request
		country_trigger = { NOT = { is_neighbor_of = FROM } }
		value = -50
	}
}

zero_focus_on_non_neighbour_fronts_if_losing_war = {
	enable = {
		is_in_faction = yes
		surrender_progress > 0.3
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request
		country_trigger = { NOT = { is_neighbor_of = FROM } }
		value = -100
	}
}

dont_protect_other_continent_allies_if_war_at_home_europe = {
	allowed = {
		is_in_europe = yes
	}
	enable = {
		is_in_faction = yes
		has_war = yes
		any_home_area_neighbor_country = {
			has_war_with = ROOT
			has_capitulated = no
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_unit_request
		area = global
		value = -90
	}
	ai_strategy = {
		type = front_unit_request
		area = western_europe #counteracts globally above
		value = 90
	}
	ai_strategy = {
		type = front_unit_request
		area = eastern_europe #counteracts globally above
		value = 90
	}
	ai_strategy = {
		type = front_unit_request
		area = scandinavia #counteracts globally above
		value = 90
	}
	ai_strategy = {
		type = front_unit_request
		area = balkans #counteracts globally above
		value = 90
	}
}
