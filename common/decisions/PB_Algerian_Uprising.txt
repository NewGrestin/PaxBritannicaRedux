PB_Algeria_Uprising = {

	PB_Algerian_Forced_Peace = {
		icon = generic_operation

		allowed = {
			original_tag = ALG
		    original_tag = FRG
		    original_tag = BAF
		}

		available = {
			OR = {
				has_global_flag = PB_Algeria_Peace
				AND = {
				    ALG = {
					    exists = no
				    }
				    FRG = {
					    exists = no
				    }
			    }
			}
		}

		fire_only_once = yes

		is_good = yes

		ai_will_do = {
			factor = 10
		}

		days_mission_timeout = 100

		timeout_effect = {
			BAF = {
				white_peace = ALG
				white_peace = FRG
			}
			hidden_effect = {
				BAF = {
					news_event = { 
						id = PBNews.191
					    days = 1 
					} 
				}
				set_global_flag = PB_Algeria_Peace
				if = {
					limit = {
						ALG = {
							exists = yes
						}
					}
					ALG = {
					    country_event = { id = magman.3 days = 5 }
				    }
					BAF = { 
						every_owned_state = { 
							limit = { is_algeria = yes }
							transfer_state_to = ALG
						}
					}
				}
				else = {
					BAF = { 
						every_owned_state = { 
							limit = { is_algeria = yes }
							transfer_state_to = FRG
						}
					}
					FRA = {
			            country_event = { id = magman.5 days = 3 }
		            }
				}
			}
		}
	}
}
