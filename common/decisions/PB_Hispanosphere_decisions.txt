    HSP_war_completion_countdown = { 
        allowed = {
            always = no
        }
        
		available = {
			hidden_trigger = { always = no }
		}

        selectable_mission = no
        
        is_good = no

        fire_only_once = yes

        days_mission_timeout = 250
        timeout_effect = {
            set_temp_variable = { chr_cap = CHR.surrender_progress }
            set_temp_variable = { col_cap = COL.surrender_progress }

            if = {
                limit = { check_variable = { chr_cap > col_cap } }
                COL = { 
                    start_peace_conference = {
                        tag = CHR
                        score_factor = 1
                        winner_scope = ALL
                        loser_scope = FACTION
                    }
                }
            }
            else = {
                CHR = { 
                    start_peace_conference = {
                        tag = COL
                        score_factor = 1
                        winner_scope = ALL
                        loser_scope = FACTION
                    }
                }
                COL = { 
                    declare_war_on = {
                        target = CHR
                        type = annex_everything
                    }
                }
            }
        }
    }

            }

                COL = { transfer_state = 305 }

                limit = {
                    check_variable = {
                        var = CHR.COL_influence

                        compare = greater_than
                    }
