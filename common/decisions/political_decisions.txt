	ban_ni_party = {

		icon = generic_civil_support

		available = {
			ni < 0.15
		}

		visible = {
			NOT = { 
				has_government = ni 
				has_authoritarian_ideology_group = yes
			}
			ni > 0
		}

		remove_trigger = {
			hidden_trigger = {
				has_government = ni
			}
		}
	
		modifier = {
			ni_drift = -0.05
		}
	
		days_remove = -1
		cost = 100
		
		complete_effect = {
			hidden_effect = {
				set_temp_variable = {
					var = ban_ideology_var
					value = party_popularity@ni
				}
				multiply_temp_variable = {
					var = ban_ideology_var
					value = -1
				}
				add_popularity = {
					ideology = ni
					popularity = ban_ideology_var
				}
				multiply_temp_variable = {
					var = ban_ideology_var
					value = 0.66
				}
			}
			ban_party_ni = yes
			add_stability = ban_ideology_var
		}
	}
