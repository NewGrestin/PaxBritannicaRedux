on_actions = {  
    on_civil_war_end = {
        effect = {
            # Jacobinist victory
            if = {
                limit = {
                    original_tag = HUN
                    has_government = jc
                    NOT = { HWA = { exists = yes } }
                }
                HUN = { 
                    news_event = { id = PBNews.363 days = 1 }
                    set_capital = { state = 43 }
					add_ideas = { 
						civilian_economy
						volunteer_only
					}
					
                }
            }
            # Reactionary victory
            if = {
                limit = {
                    original_tag = HWA
                    has_government = ds
                    NOT = { HUN = { exists = yes } }
                }
                HWA = { 
                    news_event = { id = PBNews.362 days = 1 }
                    set_cosmetic_tag = HUN_Monarchy_Restored
                    set_capital = { state = 43 }
					add_ideas = { 
						civilian_economy
						volunteer_only
					}
                }
            }
        }   
    }

    on_weekly_HUN = {
		effect = {
			set_variable = { HUN_revisionist_club_influence = party_popularity@ds }
			multiply_variable = {
				var = HUN_revisionist_club_influence
				value = 100
			}
			set_variable = { HUN_civic_radical_club_influence = party_popularity@ss }
			multiply_variable = {
				var = HUN_civic_radical_club_influence
				value = 100
			}
			set_variable = { HUN_independent_and_48s_club_influence = party_popularity@lb }
			multiply_variable = {
				var = HUN_independent_and_48s_club_influence
				value = 100
			}
			set_variable = { HUN_galilei_club_influence = party_popularity@jc }
			multiply_variable = {
				var = HUN_galilei_club_influence
				value = 100
			}
			set_variable = { HUN_party_influence = 0 }
            set_variable = { HUN_neutral_influence = 0 }
            set_variable = { HUN_opposition_influence = 0 }
			# Technocrats
			if = {
				limit = { 
					OR = {
						check_variable = { HUN_technocrats_loyalty = 70 compare = greater_than_or_equals }
						has_country_flag = HUN_technocrats_integrated
					}
				}
				add_to_variable = { HUN_party_influence = HUN_revisionist_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_technocrats_loyalty = -10 compare = greater_than_or_equals } }
				add_to_variable = { HUN_neutral_influence = HUN_revisionist_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_technocrats_loyalty = -10 compare = less_than } }
				add_to_variable = { HUN_opposition_influence = HUN_revisionist_club_influence }
			}
			#Civic Radicals
			if = {
				limit = { 
					OR = {
						check_variable = { HUN_civic_radical_club_loyalty = 70 compare = greater_than_or_equals }
						has_country_flag = HUN_civic_radicals_integrated
					}
				}
				add_to_variable = { HUN_party_influence = HUN_civic_radical_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_civic_radical_club_loyalty = -10 compare = greater_than_or_equals } }
				add_to_variable = { HUN_neutral_influence = HUN_civic_radical_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_civic_radical_club_loyalty = -10 compare = less_than } }
				add_to_variable = { HUN_opposition_influence = HUN_civic_radical_club_influence }
			}
			#48s Club
			if = {
				limit = { 
					OR = {
						check_variable = { HUN_independent_and_48s_club_loyalty = 70 compare = greater_than_or_equals }
						has_country_flag = HUN_48s_integrated
					}
				}
				add_to_variable = { HUN_party_influence = HUN_independent_and_48s_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_independent_and_48s_club_loyalty = -10 compare = greater_than_or_equals } }
				add_to_variable = { HUN_neutral_influence = HUN_independent_and_48s_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_independent_and_48s_club_loyalty = -10 compare = less_than } }
				add_to_variable = { HUN_opposition_influence = HUN_independent_and_48s_club_influence }
			}
			# Galilei
			if = {
				limit = { 
					OR = {
						check_variable = { HUN_galilei_club_loyalty = 70 compare = greater_than_or_equals }
						has_country_flag = HUN_galilei_club_integrated
					}
				}
				add_to_variable = { HUN_party_influence = HUN_galilei_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_galilei_club_loyalty = -10 compare = greater_than_or_equals } }
				add_to_variable = { HUN_neutral_influence = HUN_galilei_club_influence }
			}
			else_if = {
				limit = { check_variable = { HUN_galilei_club_loyalty = -10 compare = less_than } }
				add_to_variable = { HUN_opposition_influence = HUN_galilei_club_influence }
			}

            # Crab Mentality Change

            if = {
                limit = {
                    AND = {
                        OR = {
                            check_variable = { HUN_party_influence = HUN_neutral_influence compare = greater_than_or_equals }
                            check_variable = { HUN_party_influence = HUN_neutral_influence compare = less_than }
                        }
                        check_variable = { HUN_party_influence = HUN_opposition_influence compare = less_than }
                    }
					NOT = { has_idea = PB_HUN_Crab_Mentality_crumbling }
                }
                country_event = {
					id = hungary.9
				}
            }
            else_if = {
                limit = {
                    AND = {
                        check_variable = { HUN_party_influence = HUN_neutral_influence compare = less_than }
                        check_variable = { HUN_party_influence = HUN_opposition_influence compare = greater_than_or_equals }
                    }
					NOT = { has_idea = PB_HUN_Crab_Mentality }
                }
                country_event = {
					id = hungary.10
				}
            }
            else_if = {
                limit = {
                    AND = {
                        check_variable = { HUN_party_influence = HUN_neutral_influence compare = greater_than_or_equals }
                        check_variable = { HUN_party_influence = HUN_opposition_influence compare = greater_than_or_equals }
                    }
					NOT = { has_idea = PB_HUN_Crab_Mentality_stable }
                }
                country_event = {
					id = hungary.11
				}
            }
		}
	}
}
