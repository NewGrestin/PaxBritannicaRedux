on_actions = {  
    on_civil_war_end = {
        effect = {
            # Jacobinist victory
            if = {
                limit = {
                    original_tag = HUN
                    has_government = jc
                    NOT = { HWA = { exists = yes } }
                }
                HUN = { 
                    news_event = { id = PBNews.363 days = 1 }
                    set_capital = { state = 43 }
					add_ideas = { 
						civilian_economy
						volunteer_only
					}
					
                }
            }
            # Reactionary victory
            if = {
                limit = {
                    original_tag = HWA
                    has_government = ds
                    NOT = { HUN = { exists = yes } }
                }
                HWA = { 
                    news_event = { id = PBNews.362 days = 1 }
                    set_cosmetic_tag = HUN_Monarchy_Restored
                    set_capital = { state = 43 }
					add_ideas = { 
						civilian_economy
						volunteer_only
					}
                }
            }
        }   
    }
}
