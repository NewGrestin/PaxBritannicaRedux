on_actions = {
	on_assume_faction_leadership = {
		effect = {
			if = {
				limit = { FROM = { tag = IMP } }
				save_global_event_target_as = IMP_leader
			}

			else_if = {
				limit = { FROM = { tag = NMX } }
				save_global_event_target_as = NMX_leader
			}

			else_if = {
				limit = { FROM = { tag = SMX } }
				save_global_event_target_as = SMX_leader
			}
		}
	}
}
