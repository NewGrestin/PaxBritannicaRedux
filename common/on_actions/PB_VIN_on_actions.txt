on_actions = {
	on_startup = {
		effect = {
			VIN = { country_event = { id = vietnam_associates.1 days = 10 random_days = 10 }
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					OR = {
						original_tag = VIN
						is_in_faction_with = VIN
					}
					VIN = {
						NOT = { has_country_flag = VIN_Finished_War }
						owns_state = 840 #Tonkin
					}
					FROM = { tag = BFE }
					FROM.FROM = {
						state = 1119
					}
				}
				VIN = {
					set_country_flag = VIN_Finished_War

					transfer_state = 1119
					transfer_state = 286
					transfer_state = 671
					set_autonomy = {
						target = CAM
						autonomous_state = autonomy_PB_Sister_Republic
					}
					set_autonomy = {
						target = LAO
						autonomous_state = autonomy_PB_Sister_Republic
					}
					country_event = {
						id = vietnam.1
						days = 14
					}
					news_event = {
						id = PBNews.32
						days = 1
					}
					white_peace = BFE
					drop_cosmetic_tag = yes
				}
				LAO = {
					transfer_state = 670
					transfer_state = 1419
					transfer_state = 1122
					transfer_state = 1418
					white_peace = BFE
				}
				CAM = {
					transfer_state = 1118
					transfer_state = 1417
					transfer_state = 741
					white_peace = BFE
					drop_cosmetic_tag = yes
				}
			}
		}
	}
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					original_tag = VIN
					AND = {
						BFE = {
							NOT = {
								has_country_flag = BFE_Finished_War
							}
						}
						country_exists = BFE
						BFE = {
							controls_state = 840
						}
					}
				}
				BFE = {
					set_country_flag = BFE_Finished_War
					if = {
						limit = {
							country_exists = CAM
						}
						white_peace = CAM
						annex_country = {
							target = CAM
							transfer_troops = no
						}
					}
				}
				news_event = {
					id = PBNews.31
					days = 1
				}
			}
		}
	}
}
