						target = ROOT
						modifier = PB_same_ideology
					}
					remove_opinion_modifier = {
						target = ROOT
						modifier = PB_similar_ideology
					}
					remove_opinion_modifier = {
						target = ROOT
						modifier = PB_different_ideology
					}
					remove_opinion_modifier = {
						target = ROOT
						modifier = PB_enemy_ideology
					}
				}
				if = {
					limit = { has_government = ROOT }
					add_opinion_modifier = {
						target = ROOT
						modifier = PB_same_ideology
					}
					reverse_add_opinion_modifier = {
						target = ROOT
						modifier = PB_same_ideology
					}
				}
				else_if = {
					limit = {
						OR = {
							OR = {
								AND = {
									has_authoritarian_ideology_group = yes
									PREV = { has_authoritarian_ideology_group = yes }
								}
								AND = {
									has_reformist_ideology_group = yes
									PREV = { has_reformist_ideology_group = yes }
								}
								AND = {
									has_revolutionary_ideology_group = yes
									PREV = { has_revolutionary_ideology_group = yes }
								}
							}
						}
					}
					add_opinion_modifier = {
						target = ROOT
						modifier = PB_similar_ideology
					}
					reverse_add_opinion_modifier = {
						target = ROOT
						modifier = PB_similar_ideology
					}
				}
				else_if = {
					limit = {
						OR = {
							AND = {
								has_authoritarian_ideology_group = yes
								PREV = { has_revolutionary_ideology_group = yes }
							}
							AND = {
								has_revolutionary_ideology_group = yes
								PREV = { has_authoritarian_ideology_group = yes }
							}
						}
					}
					add_opinion_modifier = {
						target = ROOT
						modifier = PB_enemy_ideology
					}
					reverse_add_opinion_modifier = {
						target = ROOT
						modifier = PB_enemy_ideology
					}
				}
				else = {
					add_opinion_modifier = {
						target = ROOT
						modifier = PB_different_ideology
					}
					reverse_add_opinion_modifier = {
						target = ROOT
						modifier = PB_different_ideology
					}
				}
			}
