	#ROOT is new controller
	#FROM is old controller
	#FROM.FROM is state ID

		effect = {
			if = {
				limit = {
					has_war_with = FROM
					FROM.FROM = {
						is_owned_by = FROM
						NOT = { has_state_flag = city_has_been_captured_once }
					}
				}
				FROM.FROM = { save_event_target_as = captured_city }
				# Koln
				if = {
					limit = {
						FROM.FROM = { state = 909 }
						var:global.province_controllers^14513 = { has_war_with = FROM }
					}
					news_event = PB_city_capture.1
				}
				#Madrid
				else_if = {
					limit = {
						FROM.FROM = { state = 41 }
						var:global.province_controllers^15578 = { has_war_with = FROM }
					}
					news_event = PB_city_capture.2
				}
				#Moscow
				else_if = {
					limit = {
						FROM.FROM = { state = 219 }
						var:global.province_controllers^6380 = { has_war_with = FROM }
					}
					news_event = PB_city_capture.3
				}
				#Paris
				else_if = {
					limit = {
						FROM.FROM = { state = 16 }
						var:global.province_controllers^11506 = { has_war_with = FROM }
					}
					news_event = PB_city_capture.4
				}
				#Finnish War Capture City Events
				else_if = { # Fall of Nyslot
					limit = {
						winter_war_active = yes

						ROOT = { tag = FIN }

						country_event = sweden_finland.1
					}
				}

					limit = {
						winter_war_active = yes
						FROM.FROM = { state = 111 }
						ROOT = { tag = FIN }
						var:global.province_controllers^15804 = { has_war_with = FROM }
					}
					SWE = {
						country_event = sweden_finland.2
					}
				}
			}
		}
	}

}
