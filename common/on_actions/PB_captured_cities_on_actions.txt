on_actions = {

	#ROOT is new controller
	#FROM is old controller
	#FROM.FROM is state ID

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					has_war_with = FROM
					FROM.FROM = {
						is_owned_by = FROM
						NOT = { has_state_flag = city_has_been_captured_once }
					}
				}
				FROM.FROM = { save_event_target_as = captured_city }
				# Koln
				if = {
					limit = {
						FROM.FROM = { state = 909 }
						ROOT = { controls_province = 14513 }
					}
					news_event = PB_city_capture.1
				}
				#Madrid
				else_if = {
					limit = {
						FROM.FROM = { state = 41 }
						ROOT = { controls_province = 15578 }
					}
					news_event = PB_city_capture.2
				}
				#Moscow
				else_if = {
					limit = {
						FROM.FROM = { state = 219 }
						ROOT = { controls_province = 6380 }
					}
					news_event = PB_city_capture.3
				}
				#Paris
				else_if = {
					limit = {
						FROM.FROM = { state = 16 }
						ROOT = { controls_province = 11506 }
					}
					news_event = PB_city_capture.4
				}
			}
		}
	}

}
