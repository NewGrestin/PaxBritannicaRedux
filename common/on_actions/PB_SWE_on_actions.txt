on_actions = {
	on_startup = {
		effect = {
			SWE = {
				country_event = {
					id = sweden_flavor.21
					days = 450
				}
				country_event = {
					id = sweden_flavor.22
					days = 370
				}
				country_event = {
					id = sweden_flavor.23
					days = 470
				}
				country_event = {
					id = sweden_flavor.24
					days = 560
				}
				country_event = {
					id = sweden_flavor.25
					days = 970
				}
			}
		}
	}
	on_send_volunteers = {
        effect = {
            if = {
                limit = {
					ROOT = { tag = SWE }
                    FROM = { 
						NOT = { has_country_flag = SWE_already_sent_volunteers }
						any_enemy_country = {
							OR = {
								tag = RUS
								has_country_flag = Tag_in_Russian_Sphere
							}
						}
					}
                }
				FROM = { set_country_flag = SWE_already_sent_volunteers }
				ROOT = {
					random_list = {
						1 = { 
							set_variable = { SWE_hedin_experience_gain_army_factor = 0.05 }
							set_variable = { SWE_hedin_mountaineers_factor = 0.10 }
							set_variable = { SWE_hedin_acclimatization_factor = 0.15 }
						}
						1 = { 
							set_variable = { SWE_hedin_acclimatization_factor = 0.15 }
							set_variable = { SWE_hedin_winter_attrition_factor = -0.05 }
							set_variable = { SWE_hedin_morale_factor = 0.03 }
						}
						1 = { 
							set_variable = { SWE_hedin_supply_consumption_factor = -0.03 }
							set_variable = { SWE_hedin_reinforce_rate = 0.01 }
							set_variable = { SWE_hedin_recon_factor = 0.05 }
						}
						1 = { 
							set_variable = { SWE_hedin_experience_gain_army_factor = 0.05 }
							set_variable = { SWE_hedin_sf_training_factor = -0.05 }
							set_variable = { SWE_hedin_mountaineers_factor = 0.10 }
						}
					}
				}
            }
        }
    }
}
