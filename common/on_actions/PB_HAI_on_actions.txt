on_actions = {
	on_startup = {
		effect = {
			HAI = {
				country_event = {
					id = haiti.1
					days = 50
				}
			}
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					tag = HAI
                    FROM = { tag = COL }
                    FROM.FROM = {
                        OR = {
                            state = 319
							state = 306
                        }
                    }
					NOT = {
						HAI = {
							has_country_flag = HAI_Finished_War
						}
					}
				}
				HAI = {
					set_country_flag = HAI_Finished_War
					transfer_state = 319
					country_event = {
						id = haiti_warfare.3
						days = 1
					}
					white_peace = COL
				}
				COL = {
					white_peace = HAI
				}
			}
		}
	}
}
