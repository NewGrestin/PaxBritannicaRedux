on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = {
					tag = DEN
					has_global_flag = DEN_agenda_active
				}
				add_to_variable = { global.DEN_Agenda_Timer = -1 }
				if = {
					limit = { check_variable = { global.DEN_Agenda_Timer < 1 } }
					DEN_fail_agenda = yes
				}
			}
		}
	}
}
