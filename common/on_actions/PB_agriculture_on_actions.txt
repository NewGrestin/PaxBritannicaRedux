on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { grain < 1 }
				random_owned_controlled_state = {
					add_resource = {
						type = grain
    					amount = 1
					}
				}
			}
		}
		effect = {
			every_country = {
				PB_calculate_grain_fulfillment = yes
				add_dynamic_modifier = { modifier = PB_agriculture_consumption }
				set_country_flag = PB_agriculture_has_started_calculating_flag
			}
		}
	}
	on_weekly = {
		effect = {
			PB_calculate_grain_fulfillment = yes
		}
	}
}
