	PB_social_development_menu_open = {
		window_name = "PB_social_development_menu_open"
		context_type = player_context
		parent_window_name = countrypoliticsview

		effects = {
			PB_social_development_menu_open_button_click = {
				if = {
					limit = { has_country_flag = PB_social_development_menu_open }
					clr_country_flag = PB_social_development_menu_open
				}
				else = {
					set_country_flag = PB_social_development_menu_open
				}
			}
		}

		ai_enabled = { always = yes }
	}

	PB_social_development_menu = {
		window_name = "PB_social_development_menu"
		context_type = player_context
		parent_window_name = "countrypoliticsview_anchor"
		visible = { has_country_flag = PB_social_development_menu_open }

		effects = {
			PB_social_development_menu_close_button_click = {
				if = {
					limit = { has_country_flag = PB_social_development_menu_open }
					clr_country_flag = PB_social_development_menu_open
				}
			}
		}

		triggers = {
			PB_SOCDEV_POV_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_POV_count < 0

			PB_SOCDEV_IND_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_IND_count < 0
				}
			}
			PB_SOCDEV_LIT_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_LIT_count < 0
				}
			}
			PB_SOCDEV_ECON_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_ECON_count < 0
				}
			}
			PB_SOCDEV_MIL_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_MIL_count < 0
				}
			}
			PB_SOCDEV_CIV_progress_bar_bad_visible = {
				check_variable = {
					SOCDEV_CIV_count < 0
				}
			}
		}

		properties = {
			PB_SOCDEV_POV_progress_bar = {
				frame = SOCDEV_POV_count
			}
			PB_SOCDEV_IND_progress_bar = {
				frame = SOCDEV_IND_count
			}
			PB_SOCDEV_LIT_progress_bar = {

	PB_social_development_map_mode_toggle = {
		window_name = "PB_social_development_map_mode_toggle"
		context_type = player_context
		dirty = selected_social_development_map_mode_toggle
		mapmode = social_development_map_mode

		triggers = {
			toggle_social_development_map_mode_button_click_enabled = {
				check_variable = { var = selected_social_development_map_mode_toggle value = current_toggle_socdev compare = not_equals }
			}
			temp_array_calculator_click_enabled = {
				resize_temp_array = { social_development_map_mode_toggles_temp_array = @NUM_OF_SOCDEV_MAP_MODE_TYPES }
			}
		}

		properties = {
			toggle_social_development_map_mode_button = {
				image = "GFX_social_development_map_mode_toggle_[?current_toggle_socdev]"
			}
		}

		effects = {
			toggle_social_development_map_mode_button_click = {
				set_variable = { selected_social_development_map_mode_toggle = current_toggle_socdev }
				force_update_map_mode = {
					mapmode = social_development_map_mode
				}
			}
		}

		dynamic_lists = {
			social_development_map_mode_toggles_gridbox = {
				entry_container = "social_development_map_mode_toggles_entry"
				array = social_development_map_mode_toggles_temp_array
				index = current_toggle_socdev
			}
		}

		ai_enabled = { always = no }
	}
