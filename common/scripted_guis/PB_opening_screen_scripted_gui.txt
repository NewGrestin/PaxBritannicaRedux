scripted_gui = {
	PB_opening_screen = {
		context_type = player_context
		window_name = "PB_opening_screen_container"
		dirty = PB_opening_screen_reload

		ai_enabled = { always = no }

		visible = {
			has_variable = PB_opening_screen_var
		}

		triggers = {
			country_lore_button_click_enabled = { NOT = { check_variable = { PB_opening_screen_var = 1 } } }
			country_paths_button_click_enabled = { NOT = { check_variable = { PB_opening_screen_var = 2 } } }
			politics_button_click_enabled = { NOT = { check_variable = { PB_opening_screen_var = 3 } } }
			world_lore_button_enabled = { NOT = { check_variable = { PB_opening_screen_var = 4 } } }
			credits_button_enabled = { NOT = { check_variable = { PB_opening_screen_var = 5 } } }
		}

		effects = {
			country_lore_button_click = {
				set_variable = { PB_opening_screen_var = 1 }
				set_variable_to_random = PB_opening_screen_reload
			}
			country_paths_button_click = {
				set_variable = { PB_opening_screen_var = 2 }
				set_variable_to_random = PB_opening_screen_reload
			}
			politics_button_click = {
				set_variable = { PB_opening_screen_var = 3 }
				set_variable_to_random = PB_opening_screen_reload
			}
			world_lore_button_click = {
				set_variable = { PB_opening_screen_var = 4 }
				set_variable_to_random = PB_opening_screen_reload
			}
			credits_button_click = {
				set_variable = { PB_opening_screen_var = 5 }
				set_variable_to_random = PB_opening_screen_reload
			}

			continue_button_click = {
				clear_variable = PB_opening_screen_var
				set_variable_to_random = PB_opening_screen_reload
			}
		}

		properties = {
			PB_opening_screen_country_picture = {
				image = "[ROOT.GetCountryOpeningPicture]"
			}
		}
	}

	PB_opening_screen_country_lore_tab = {
		context_type = player_context
		window_name = PB_opening_screen_country_lore_container
