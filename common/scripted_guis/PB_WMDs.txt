scripted_gui = {
	PB_WMD_menu_button_scripted_gui = {
		window_name = "PB_WMD_menu_button_container"
		context_type = selected_state_context
		parent_window_token = selected_state_view

		triggers = {
			PB_Chemical_Weapons_Deployment_Icon_visible = {
				ROOT = {
					OR = {
						has_tech = chemweapons_basic_irritant
						has_tech = chemweapons_basic_choking_agent
						has_tech = chemweapons_basic_psychadelic
					}
					NOT = {
						has_idea = chemical_weapons_cooldown
					}
				}
				controller = {
					has_war_with = ROOT
				}
				NOT = {
					has_any_WMD_related_modifier = yes
				}
			}

			PB_Biological_Weapons_Deployment_Icon_visible = {
				ROOT = {
					OR = {
						has_tech = bioweapons_anthrax
						has_tech = bioweapons_smallpox
						has_tech = bioweapons_botulism
						has_tech = bioweapons_Influenza
					}
					NOT = {
						has_idea = biological_weapons_cooldown
					}
				}
				controller = {
					has_war_with = ROOT
				}
				NOT = {
					OR = {
						has_global_flag = PB_Global_Pandemic_Crisis_Initiated
						has_any_WMD_related_modifier = yes
					}
				}
			}

			PB_Livewire_Weapons_Deployment_Icon_visible = {
				ROOT = {
					OR = {
						has_tech = wmd_livewire
					}
					NOT = {
						has_idea = livewire_weapons_cooldown
					}
				}
				controller = {
					has_war_with = ROOT
				}
				NOT = {
					OR = {
						has_global_flag = PB_Global_Pandemic_Crisis_Initiated
						has_any_WMD_related_modifier = yes
					}
				}
			}
		}

		effects = {
			PB_Chemical_Weapons_Deployment_Icon_click = {
				deploy_chemical_weapons = yes

				controller = {
					country_event = {
						id = chembio.4
						days = 1
					}					
				}
				add_to_variable = {
					global.WMD_update = 1
				}
			}
			PB_Biological_Weapons_Deployment_Icon_click = {
				deploy_biological_weapons = yes

				controller = {
					country_event = {
						id = chembio.3
						days = 1
					}					
				}
				add_to_variable = {
					global.WMD_update = 1
				}

				if = {
					limit = {
						NOT = {
							has_global_flag = PB_Global_Bioweapons_Deployed_Initial
						}
					}
					set_global_flag = PB_Global_Bioweapons_Deployed_Initial
				}
			}

			PB_Livewire_Weapons_Deployment_Icon_click = {
				deploy_livewire_weapons = yes
				controller = {
					country_event = {
						id = chembio.3
						days = 1
					}					
				}
				add_to_variable = {
					global.WMD_update = 1
				}
			}
		}

		ai_enabled = { always = no }
	}
}
