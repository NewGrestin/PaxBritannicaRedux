## Civilian population
# Gets the scale (in thousands/millions)
defined_text = {
	name = PB_agriculture_get_pop_scale
	text = {
		trigger = {
			check_variable = { max_manpower_k < 1000 }
		}
		localization_key = "PB_agriculture_pop_thousands"
	}
	text = {
		trigger = {
			set_temp_variable = { manpower_in_millions = max_manpower_k }
			divide_temp_variable = { manpower_in_millions = 1000 }
		}
		localization_key = "PB_agriculture_pop_millions"
	}
}

# Get multiplicative factor in millions
defined_text = {
	name = PB_agriculture_get_pop_mult_scale
	text = {
		trigger = {
			set_temp_variable = { manpower_mult_in_millions = PB_agriculture_per_pop }
			multiply_temp_variable = { manpower_mult_in_millions = 1000 }
		}
		localization_key = "PB_agriculture_pop_mult_millions"
	}
}

# Get the severity of the famine
defined_text = {
	name = PB_agriculture_get_famine_severity
	text = {
		trigger = {
			check_variable = { var = PB_agriculture_fulfillment value = 10 compare = less_than_or_equals }
		}
		localization_key = "PB_agriculture_famine_3"
	}
	text = {
		trigger = {
			check_variable = { var = PB_agriculture_fulfillment value = 30 compare = less_than_or_equals }
		}
		localization_key = "PB_agriculture_famine_2"
	}
	text = {
		localization_key = "PB_agriculture_famine_1"
	}
}

## Intel ledger

defined_text = {
	name = PB_agriculture_get_intel_ledger_text
	text = {
		trigger = {
			OR = {
				ROOT = {
					check_variable = { var = civilian_intel@PREV value = 50 compare = greater_than_or_equals }
				}
				is_in_faction_with = ROOT
				is_subject_of = ROOT
				has_subject = ROOT
				original_tag = ROOT
			}
		}
		localization_key = PB_agriculture_intel_full
	}
	text = {
		trigger = {
			ROOT = {
				check_variable = { var = civilian_intel@PREV value = 10 compare = greater_than_or_equals }
			}
		}
		localization_key = PB_agriculture_intel_partial
	}

}
