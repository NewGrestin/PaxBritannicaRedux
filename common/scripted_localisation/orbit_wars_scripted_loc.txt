defined_text = {
	name = OW_get_top_text

	# VICTORY STATES — MUST COME FIRST

	text = {
		trigger = {
			has_global_flag = orbit_wars_victor_@ENG
		}
		localization_key = OW_ENG_orbital_victor
	}

	text = {
		trigger = {
			has_global_flag = orbit_wars_victor_@RUS
		}
		localization_key = OW_RUS_orbital_victor
	}

	# NORMAL ORBITAL SUPERIORITY

	text = {
		trigger = {
			OR = {
				AND = {
					NOT = { ENG = { has_country_flag = has_satellite_in_orbit } }
					NOT = { RUS = { has_country_flag = has_satellite_in_orbit } }
				}
				AND = {
					ENG = { has_country_flag = has_satellite_in_orbit }
					RUS = { has_country_flag = has_satellite_in_orbit }
				}
			}
			NOT = { has_global_flag = orbit_wars_victor_@ENG }
			NOT = { has_global_flag = orbit_wars_victor_@RUS }
		}
		localization_key = OW_no_orbital_superiority
	}

	text = {
		trigger = {
			ENG = { has_country_flag = has_satellite_in_orbit }
			NOT = { RUS = { has_country_flag = has_satellite_in_orbit } }
		}
		localization_key = OW_ENG_orbital_superiority
	}

	text = {
		trigger = {
			NOT = { ENG = { has_country_flag = has_satellite_in_orbit } }
			RUS = { has_country_flag = has_satellite_in_orbit }
		}
		localization_key = OW_RUS_orbital_superiority
	}
}
