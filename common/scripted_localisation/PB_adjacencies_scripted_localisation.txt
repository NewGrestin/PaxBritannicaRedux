# Tooltip for nicaraguan canal
defined_text = {
	name = GetNicaraguaCanalBlockText
	text = {
		trigger = {
			NOT = { has_global_flag = NICARAGUA_CANAL_BUILT }
		}
		localization_key = nicaragua_canal_is_during_construction_tt
	}
	text = {
		trigger = {
			has_global_flag = NICARAGUA_CANAL_BLOCKED
		}
		localization_key = nicaragua_canal_is_blocked_tt
	}
	text = {
		trigger = {
			has_country_flag = NICARAGUA_CANAL_BLOCKED_FOR_COUNTRY
		}
		localization_key = nicaragua_canal_is_blocked_for_country_tt
	}
}
