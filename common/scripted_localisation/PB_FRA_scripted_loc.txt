### FERVOUR ###
defined_text = { 
    name = get_FRA_Nationalistic_Fervour_Loc
    
    text = { 
        trigger = {
            check_variable = { var = FRA_fervour_var value = 75 compare = greater_than_or_equals }
        }
        localization_key = FRA_high_fervour_tt
    }
    text = { 
        trigger = {
            AND = {
                check_variable = { FRA_fervour_var < 75 }
                check_variable = { FRA_fervour_var > 50 }
            }
        }
        localization_key = FRA_medium_fervour_tt
    }
    text = { 
        trigger = {
            check_variable = { var = FRA_fervour_var value = 50 compare = less_than_or_equals }
        }
        localization_key = FRA_low_fervour_tt
    }

    
}
