defined_text = {
	name = SWE_get_riksdag_chance

	text = {
		trigger = {
			check_variable = { SWE_riksdag_support < 50 }
		}
		localization_key = "SWE_riksdag_chance_0_text"
	}
	text = {
		trigger = {
			check_variable = { SWE_riksdag_support > 49 }
			check_variable = { SWE_riksdag_support < 60 }
		}
		localization_key = "SWE_riksdag_chance_1_text"
	}
	text = {
		trigger = {
			check_variable = { SWE_riksdag_support > 59 }
			check_variable = { SWE_riksdag_support < 75 }
		}
		localization_key = "SWE_riksdag_chance_2_text"
	}
	text = {
		trigger = {
			check_variable = { SWE_riksdag_support > 74 }
			check_variable = { SWE_riksdag_support < 90 }
		}
		localization_key = "SWE_riksdag_chance_3_text"
	}
	text = {
		trigger = {
			check_variable = { SWE_riksdag_support > 89 }
		}
		localization_key = "SWE_riksdag_chance_4_text"
	}
}
