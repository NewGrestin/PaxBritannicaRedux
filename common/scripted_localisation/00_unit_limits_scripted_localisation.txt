#recursive scripted loc
defined_text = {
	name = unit_limit_subject_bonus_breakdown
	text = {
		trigger = {
			NOT = { check_variable = { index < subjects^num } } #end loop if past final index
			set_temp_variable = { index = 0 } #for other loops down the line
		}
		localization_key = loc_empty
	}
	text = {
		trigger = { #modifier present, list scope
			var:subjects^index = {
				NOT = { check_variable = { divisions_from_subject = 0 } }
				set_temp_variable = { scope = THIS }
			}
			add_to_temp_variable = { index = 1 }
		}
		localization_key = unit_limit_subject_bonus_breakdown_desc
	}
	text = { #no modifier, move to next index
		trigger = {
			add_to_temp_variable = { index = 1 }
		}
		localization_key = "[unit_limit_subject_bonus_breakdown]"
	}
}

defined_text = {
	name = unit_limit_economy_law
	text = {
		trigger = {
			NOT = { check_variable = { modifier@unit_limit_law_bonus = 0 } }
			if = {
				limit = { has_idea = undisturbed_isolation }
				set_temp_variable = { law = token:undisturbed_isolation }
			}
			else_if = {
				limit = { has_idea = isolation }
				set_temp_variable = { law = token:isolation }
			}
			else_if = {
				limit = { has_idea = civilian_economy }
				set_temp_variable = { law = token:civilian_economy }
			}
			else_if = {
				limit = { has_idea = low_economic_mobilisation }
				set_temp_variable = { law = token:low_economic_mobilisation }
			}
			else_if = {
				limit = { has_idea = partial_economic_mobilisation }
				set_temp_variable = { law = token:partial_economic_mobilisation }
			}
			else_if = {
				limit = { has_idea = war_economy }
				set_temp_variable = { law = token:war_economy }
			}
			else_if = {
				limit = { has_idea = tot_economic_mobilisation }
				set_temp_variable = { law = token:tot_economic_mobilisation }
			}
		}
		localization_key = unit_limit_economy_law_desc
	}
}
