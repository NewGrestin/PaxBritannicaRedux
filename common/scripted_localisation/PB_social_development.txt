defined_text = {
	name = get_SD_POV_growth_type
	text = {
		trigger = {
			check_variable = { SOCDEV_POV_real_change = 0 }
		}
		localization_key = socdev_stagnant_growth
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change > 0 }
				check_variable = { SOCDEV_POV_real_change < 1 }
				NOT = {check_variable = { SOCDEV_POV_real_change = 0 } }
			}
		}
		localization_key = socdev_slow_progress
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change > 0.99 }
				check_variable = { SOCDEV_POV_real_change < 2.5 }
				NOT = {check_variable = { SOCDEV_POV_real_change = 0 } }
			}
		}
		localization_key = socdev_moderate_progress
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change > 2.49 }
				NOT = {check_variable = { SOCDEV_POV_real_change = 0 } }
			}
		}
		localization_key = socdev_rapid_progress
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change < 0 }
				check_variable = { SOCDEV_POV_real_change > 1 }
			}
		}
		localization_key = socdev_slow_regress
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change < -0.99 }
				check_variable = { SOCDEV_POV_real_change > -2.5 }
			}
		}
		localization_key = socdev_moderate_regress
	}
	text = {
		trigger = {
			AND = {
				check_variable = { SOCDEV_POV_real_change < -2.49 }
			}
		}
		localization_key = socdev_rapid_regress
	}
}
