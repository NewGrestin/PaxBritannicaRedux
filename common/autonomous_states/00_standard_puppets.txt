### PUPPETS FOR AN/CN/RC/LB/SS governments ###
## INTEGRATED PUPPET ##
autonomy_state = {
	id = autonomy_integrated_puppet

	is_puppet = yes

	min_freedom_level = 0.0

	manpower_influence = 1.0

	rule = {
		desc = RULE_DESC_IS_A_SUBJECT

		can_not_declare_war = yes
		can_decline_call_to_war = no
		can_be_spymaster = no
		contributes_operatives = no
		can_create_collaboration_government = no
	}

	modifier = {
		autonomy_manpower_share = 1.0
		can_master_build_for_us = 1
		extra_trade_to_overlord_factor = 1.0
		overlord_trade_cost_factor = -0.9
		research_sharing_per_country_bonus_factor = -0.5
		cic_to_overlord_factor = 0.25
		mic_to_overlord_factor = 0.65
		peace_score_ratio_transferred_to_overlord = 0.2
		faction_influence_contribution_factor = -0.75
	}

	ai_subject_wants_higher = {
		factor = 1.0
	}

	ai_overlord_wants_lower = {
		factor = 0.0
	}

	allowed = {
		OVERLORD = {
			NOT = {
				OR = {
					has_government = ni
					has_government = ds
					has_government = jc
				}
			}
		}
	}

	allowed_levels_filter = {
		autonomy_integrated_puppet
		autonomy_dependent_puppet
		autonomy_default_puppet
		autonomy_autonomous_puppet
		autonomy_nominal_puppet
	}

	can_take_level = {
		custom_override_tooltip = {
			tooltip = can_raise_autonomy_level
			has_country_flag = PB_can_raise_autonomy_level
		}
	}

	can_lose_level = {
		custom_override_tooltip = {
			tooltip = puppet_can_be_annexed
			always = no
		}
	}
}
