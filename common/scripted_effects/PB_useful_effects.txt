update_custom_state_pop_growth = {
	every_state = { 
		limit = { 
			OR = {
				medical_center > 0
				has_dynamic_modifier = {
					modifier = PB_state_focus_modifiers_8 #population investments
				}
			}
		}
		set_temp_variable = { modified_pop_growth = state_population_k } 
		set_temp_variable = { custom_pop_growth = 1 }
		add_to_temp_variable = { custom_pop_growth = modifier@state_monthly_population } 
		multiply_temp_variable = { custom_pop_growth = 0.00125 } 
		multiply_temp_variable = { modified_pop_growth = custom_pop_growth }
		multiply_temp_variable = { modified_pop_growth = 1 }
		add_manpower = var:modified_pop_growth
	}
}

embargo_ROOT = {
	if = {
		limit = {
			NOT = { has_country_flag = embargoed_@ROOT }
			NOT = { tag = ROOT } #sanity check
		}
		set_country_flag = embargoed_@ROOT
		ROOT = { set_country_flag = embargoed_@PREV }
		if = {
			limit = { has_dlc = "By Blood Alone" }
			if = {
				limit = { NOT = { is_embargoing = ROOT } }
				diplomatic_relation = {
					country = ROOT
					relation = embargo
				}
			}
			if = {
				limit = { NOT = { is_embargoed_by = ROOT } }
				ROOT = {
					diplomatic_relation = {
						country = PREV
						relation = embargo
					}
				}
			}
		}
		else = {
			add_opinion_modifier = {
				modifier = embargo
				target = ROOT
			}
			reverse_add_opinion_modifier = {
				modifier = embargo
				target = ROOT
			}
		}
	}
}

# ends puppet status, removes from faction and all current wars (used for rebellions)
clear_all_relations = {
	clear_spheres = yes
	if = {
		limit = { is_subject = yes }
		overlord = {
			set_autonomy = {
				target = PREV
				autonomy_state = autonomy_free
			}
		}
	}
	if = {
		limit = { is_faction_leader = yes }
		dismantle_faction = yes
	}
	else_if = {
		limit = { is_in_faction = yes }
		if = {
			limit = { is_exile_host = yes } #THIS MUST BE DONE BEFORE LEAVING A FACTION OR ELSE EXILE GOV LEAVES FACTION TOO
			var:faction_leader = {
				for_each_scope_loop = {
					array = PREV.exiles
					become_exiled_in = { target = PREV }
				}
			}
		}
		leave_faction = yes
	}
	if = {
		limit = { num_subjects > 0 }
		every_subject_country = {
			overlord = {
				set_autonomy = {
					target = PREV
					autonomy_state = autonomy_free
				}
			}
		}
	}
	if = {
		limit = { NOT = { has_variable = keep_wars } }
		separate_peace_all_wars = yes
	}
	effect_tooltip = {
		every_other_country = {
			limit = { has_guaranteed = PREV }
			diplomatic_relation = {
				country = PREV
				relation = guarantee
				active = no
			}
		}
		every_other_country = {
			limit = { has_non_aggression_pact_with = PREV }
			diplomatic_relation = {
				country = PREV
				relation = non_aggression_pact
				active = no
			}
		}
		every_other_country = {
			limit = { has_military_access_to = PREV }
			diplomatic_relation = {
				country = PREV
				relation = military_access
				active = no
			}
		}
		every_other_country = {
			limit = { is_guaranteed_by = PREV }
			PREV = {
				diplomatic_relation = {
					country = PREV
					relation = guarantee
					active = no
				}
			}
		}
		every_other_country = {
			limit = { gives_military_access_to = PREV }
			PREV = {
				diplomatic_relation = {
					country = PREV
					relation = military_access
					active = no
				}
			}
		}
		every_other_country = {
			limit = { gives_military_access_to = PREV }
			PREV = {
				diplomatic_relation = {
					country = PREV
					relation = military_access
					active = no
				}
			}
		}
		every_other_country = {
			limit = { has_market_access_with = PREV }
			diplomatic_relation = {
				country = PREV
				relation = military_access
				active = no
			}
		}
		every_other_country = {
			limit = { PREV = { has_market_access_with = PREV } }
			PREV = {
				diplomatic_relation = {
					country = PREV
					relation = military_access
					active = no
				}
			}
		}
		if = {
			limit = { has_dlc = "By Blood Alone" }
			every_other_country = {
				limit = { is_embargoing = PREV }
				diplomatic_relation = {
					country = ROOT
					relation = embargo
					active = no
				}
			}
			every_other_country = {
				limit = { is_embargoed_by = PREV }
				PREV = {
					diplomatic_relation = {
						country = PREV
						relation = embargo
						active = no
					}
				}
			}
		}
		else = {
			every_other_country = {
				limit = { has_variable = embargoed_@PREV }
				remove_opinion_modifier = {
					modifier = embargo
					target = PREV
				}
			}
			every_other_country = {
				limit = { PREV = { has_variable = embargoed_@PREV } }
				PREV = {
					remove_opinion_modifier = {
						modifier = embargo
						target = PREV
					}
				}
			}
		}
	}
	hidden_effect = {
		every_other_country = {
			diplomatic_relation = {
				country = PREV
				relation = guarantee
				active = no
			}
			diplomatic_relation = {
				country = PREV
				relation = non_aggression_pact
				active = no
			}
			diplomatic_relation = {
				country = PREV
				relation = military_access
				active = no
			}
			diplomatic_relation = {
				country = PREV
				relation = market_access_rights
				active = no
			}
			PREV = {
				diplomatic_relation = {
					country = PREV
					relation = guarantee
					active = no
				}
				diplomatic_relation = {
					country = PREV
					relation = military_access
					active = no
				}

					country = PREV

				}

		}
	}
