
open_WMD_window = {
	ROOT = {
		if = {
			limit = { WMD_window_is_open = no }
			set_country_flag = PB_WMDs_menu_open
		}
	}
}
close_WMD_window = {
	ROOT = {
		if = {
			limit = { WMD_window_is_open = yes }
			clr_country_flag = PB_WMDs_menu_open
		}
	}
}
