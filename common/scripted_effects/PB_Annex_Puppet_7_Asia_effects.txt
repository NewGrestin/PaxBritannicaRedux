setup_custom_puppet_JAP = {
	if = { #RUSSIA
		limit = { tag = RUS }
		if = {  #THEOCRATIC
			limit = {
				has_government = ds
			}
			set_autonomy = {
				target = var:tag_to_release
				autonomous_state = autonomy_PB_Russian_Theocracy
				freedom_level = 0.50
			}
			var:tag_to_release = {
				set_politics = {
					ruling_party = ds
				}
				set_party_name = {
					ideology = ds
					long_name = "Metropolitan Patriarch of Japan"
					name = "Metropolitan Patriarch of Japan"
				}
				kill_country_leader = yes
				create_country_leader = {
					name = "Sergius of Japan"
					desc = ""
					picture = GFX_Portrait_JAP_Sergius_of_Japan
					expire = "1965.1.1"
					ideology = DS_VARIANT_09
				}
				set_cosmetic_tag = JAP_RUS_Theocracy
			}
		}
		else = {  # DESPOTIC
			ROOT = {
				set_autonomy = {
					target = NOR
					autonomous_state = autonomy_client_state
					freedom_level = 0.50
				}
			}
			NOR = {
				set_politics = {
					ruling_party = ni
				}

				add_popularity = {
					ideology = ni
					popularity = 0.3
				}

				kill_country_leader = yes
				create_country_leader = {
					name = "Russian Admirality"
					desc = "PB_Johan_Bernhard_Hjort_DESC"
					picture = "Portrait_NOR_Johan_Bernhard_Hjort.dds"
					expire = "1965.1.1"
					ideology = NI_VARIANT_01
				}
				drop_cosmetic_tag = yes
				set_cosmetic_tag = JAP_RUS_Occupied
			}
		} 
	}
}
