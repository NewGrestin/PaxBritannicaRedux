PB_calculate_grain_fulfillment = {
    set_variable = { PB_agriculture_per_pop = 0.001 }
    
    set_variable = { PB_agriculture_per_pop_military = PB_agriculture_per_pop }
    multiply_variable = { PB_agriculture_per_pop_military = 5 }

    ## Modifiers calculation
    # Modified consumption = base * (1 + consumption factor) - Military
    set_temp_variable = { mil_modifier_calc = 1 }
    add_to_temp_variable = { mil_modifier_calc = modifier@food_consumption_factor_military }
    multiply_variable = { PB_agriculture_per_pop_military = mil_modifier_calc }
    
    # Modified consumption = base * (1 + consumption factor) - Civilian
    set_temp_variable = { pop_modifier_calc = 1 }
    add_to_temp_variable = { pop_modifier_calc = modifier@food_consumption_factor }
    multiply_variable = { PB_agriculture_per_pop = pop_modifier_calc }

    ## Calculating demand
    # Civilians
    set_variable = { PB_agriculture_demand_base = max_manpower_k }
    multiply_variable = { PB_agriculture_demand_base = PB_agriculture_per_pop }

    # Military
    set_variable = { PB_agriculture_total_military_pop = deployed_army_manpower_k }
    add_to_variable = { PB_agriculture_total_military_pop = deployed_navy_manpower_k }
    add_to_variable = { PB_agriculture_total_military_pop = deployed_airforce_manpower_k }

    set_variable = { PB_agriculture_demand_military = PB_agriculture_total_military_pop }
    multiply_variable = { PB_agriculture_demand_military = PB_agriculture_per_pop_military }

    ## Sum both Civilians and Military demand
    set_variable = { PB_agriculture_demand = PB_agriculture_demand_base }
    add_to_variable = { PB_agriculture_demand = PB_agriculture_demand_military }

    # Subtract grain stockpile by amount of demand
    # set_variable = { PB_agriculture_demand_rounded = PB_agriculture_demand }
    # round_variable = PB_agriculture_demand_rounded
    # force_update_dynamic_modifier = yes

    # set_variable = { PB_grain_true = resource@grain }
    # if = {
    #     limit = {
    #         has_country_flag = PB_agriculture_has_started_calculating_flag
    #     }
    #     add_to_variable = { PB_grain_true = PB_agriculture_demand_rounded }
    # }

    set_variable = { PB_agriculture_fulfillment_pct = resource@grain }
    divide_variable = { PB_agriculture_fulfillment_pct = PB_agriculture_demand }
    set_variable = { PB_agriculture_fulfillment = PB_agriculture_fulfillment_pct }
    multiply_variable = { PB_agriculture_fulfillment = 100 }
    clamp_variable = { var = PB_agriculture_fulfillment min = 0 max = 100 }

    log = "[THIS.GetTag] - resource@grain: [?resource@grain]"
    log = "[THIS.GetTag] - PB_agriculture_demand: [?PB_agriculture_demand]"
    log = "[THIS.GetTag] - PB_agriculture_fulfillment: [?PB_agriculture_fulfillment]"
}
