PB_calculate_grain_fulfillment = {
	set_variable = { PB_agriculture_per_pop = 0.001 }
	set_variable = { PB_agriculture_per_pop_raw = 0.001 }
	
	set_variable = { PB_agriculture_per_pop_military = PB_agriculture_per_pop }
	set_variable = { PB_agriculture_per_pop_military_raw = PB_agriculture_per_pop_raw }

	multiply_variable = { PB_agriculture_per_pop_military = 10 }
	multiply_variable = { PB_agriculture_per_pop_military_raw = 10 }

	## Modifiers calculation
	# Modified consumption = base * (1 + consumption factor) - Military
	set_temp_variable = { mil_modifier_calc = 1 }
	add_to_temp_variable = { mil_modifier_calc = modifier@food_consumption_factor }
	add_to_temp_variable = { mil_modifier_calc = modifier@food_consumption_factor_military }
	multiply_variable = { PB_agriculture_per_pop_military = mil_modifier_calc }

	set_variable = { PB_food_consumption_military_total = mil_modifier_calc }
	subtract_from_variable = { PB_food_consumption_military_total = 1 }

	# RAW military consumption (NO modifiers)
	# (intentionally unchanged base * 5 already applied above)

	
	# Modified consumption = base * (1 + consumption factor) - Civilian
	set_temp_variable = { pop_modifier_calc = 1 }
	add_to_temp_variable = { pop_modifier_calc = modifier@food_consumption_factor }
	add_to_temp_variable = { pop_modifier_calc = modifier@food_consumption_factor_civilian }
	multiply_variable = { PB_agriculture_per_pop = pop_modifier_calc }

	set_variable = { PB_food_consumption_civilian_total = pop_modifier_calc }
	subtract_from_variable = { PB_food_consumption_civilian_total = 1 }

	# RAW civilian consumption already in PB_agriculture_per_pop_raw (no modifiers applied)


	## Calculating demand
	# Military
	set_variable = { PB_agriculture_total_military_pop = deployed_army_manpower_k }
	add_to_variable = { PB_agriculture_total_military_pop = deployed_navy_manpower_k }
	add_to_variable = { PB_agriculture_total_military_pop = deployed_airforce_manpower_k }

	set_variable = { amount_manpower_in_deployment_queue_k = amount_manpower_in_deployment_queue }
	divide_variable = { amount_manpower_in_deployment_queue_k = 1000 }
	add_to_variable = { PB_agriculture_total_military_pop = amount_manpower_in_deployment_queue_k }

	set_variable = { foreign_manpower_k = foreign_manpower }
	divide_variable = { foreign_manpower_k = 1000 }
	add_to_variable = { PB_agriculture_total_military_pop = foreign_manpower_k }


	# Civilians (MODIFIED)
	set_variable = { PB_agriculture_demand_base = max_manpower_k }
	subtract_from_variable = { PB_agriculture_demand_base = PB_agriculture_total_military_pop }
	multiply_variable = { PB_agriculture_demand_base = PB_agriculture_per_pop }

	set_variable = { PB_agriculture_demand_military = PB_agriculture_total_military_pop }
	multiply_variable = { PB_agriculture_demand_military = PB_agriculture_per_pop_military }


	# Civilians (RAW - NO modifiers)
	set_variable = { PB_agriculture_demand_base_raw = max_manpower_k }
	subtract_from_variable = { PB_agriculture_demand_base_raw = PB_agriculture_total_military_pop }
	multiply_variable = { PB_agriculture_demand_base_raw = PB_agriculture_per_pop_raw }

	set_variable = { PB_agriculture_demand_military_raw = PB_agriculture_total_military_pop }
	multiply_variable = { PB_agriculture_demand_military_raw = PB_agriculture_per_pop_military_raw }


	## Sum both Civilians and Military demand
	set_variable = { PB_agriculture_demand = PB_agriculture_demand_base }
	add_to_variable = { PB_agriculture_demand = PB_agriculture_demand_military }

	set_variable = { PB_agriculture_demand_raw = PB_agriculture_demand_base_raw }
	add_to_variable = { PB_agriculture_demand_raw = PB_agriculture_demand_military_raw }


	# Subtract grain stockpile by amount of demand
	set_variable = { PB_agriculture_demand_rounded = PB_agriculture_demand }
	round_variable = PB_agriculture_demand_rounded


	set_variable = { PB_grain_true = resource@grain }

	if = {
	    limit = {
	        has_country_flag = PB_agriculture_has_started_calculating_flag
	    }
	    add_to_variable = { PB_grain_true = PB_agriculture_demand_raw }
	}


	set_variable = { PB_agriculture_fulfillment_pct = PB_grain_true }
	divide_variable = { PB_agriculture_fulfillment_pct = PB_agriculture_demand }
	clamp_variable = { var = PB_agriculture_fulfillment_pct min = 0 max = 1 }

	set_variable = { PB_agriculture_fulfillment = PB_agriculture_fulfillment_pct }
	multiply_variable = { PB_agriculture_fulfillment = 100 }


	## Dynamic modifier calculation for food shortages
	if = {
		limit = {
			OR = {
				check_variable = { var = PB_agriculture_fulfillment_pct value = 0.5 compare = less_than_or_equals }
				check_variable = { var = PB_agriculture_fulfillment_pct value = 0.8 compare = greater_than_or_equals }
			}
		}
		PB_evaluate_food_modifiers = yes

		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = { modifier = PB_agriculture_famine }
				}
			}
			add_dynamic_modifier = { modifier = PB_agriculture_famine }
		}
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = PB_agriculture_famine }
		}
		PB_clear_food_modifiers = yes
		remove_dynamic_modifier = { modifier = PB_agriculture_famine }
	}


	## Intel calculations
	every_other_country = {
		limit = { is_ai = no }
		if = {
			limit = {
				# 10% <= civilian_intel@PREV < 50%
				check_variable = { var = civilian_intel@PREV value = 10 compare = greater_than_or_equals }
				check_variable = { var = civilian_intel@PREV value = 50 compare = less_than }
			}
			# x' = 0.625 - (5/4)x {10 <= x < 50} -> {50 >= x' > 0}
			set_temp_variable = { epsilon = 62.5 }
			set_temp_variable = { epsilon_calc = 1.25 }
			multiply_temp_variable = { epsilon_calc = civilian_intel@PREV }
			subtract_from_temp_variable = { epsilon = epsilon_calc }

			set_temp_variable_to_random = { var = lower_delta min = 0 max = epsilon }
			set_temp_variable_to_random = { var = upper_delta min = 0 max = epsilon }
			
			#log = "THIS: [?THIS.GetTag]"
			#log = "PREV: [?PREV.GetTag]"
			set_variable = { PB_agriculture_intel_var_lower@PREV = PREV.PB_agriculture_fulfillment }
			subtract_from_variable = { PB_agriculture_intel_var_lower@PREV = lower_delta }
			clamp_variable = { var = PB_agriculture_intel_var_lower@PREV min = 0 max = 100 }

			set_variable = { PB_agriculture_intel_var_upper@PREV = PREV.PB_agriculture_fulfillment }
			add_to_variable = { PB_agriculture_intel_var_upper@PREV = upper_delta }
			clamp_variable = { var = PB_agriculture_intel_var_upper@PREV min = 0 max = 100 }

			PB_evaluate_food_modifiers_intel = yes
		}
	}


	#log = "[THIS.GetTag] - PB_grain_true: [?PB_grain_true]"
	#log = "[THIS.GetTag] - PB_agriculture_demand: [?PB_agriculture_demand]"
	#log = "[THIS.GetTag] - PB_agriculture_demand_raw: [?PB_agriculture_demand_raw]"
	#log = "[THIS.GetTag] - PB_agriculture_fulfillment: [?PB_agriculture_fulfillment]"
}

PB_clear_food_modifiers = {
	clear_variable = PB_agriculture_dynmod_10_20_30
	clear_variable = PB_agriculture_dynmod_10_20_30_neg
	clear_variable = PB_agriculture_dynmod_5_10_20
	clear_variable = PB_agriculture_dynmod_5_10_20_neg
	clear_variable = PB_agriculture_dynmod_0_10_30
	clear_variable = PB_agriculture_dynmod_0_10_30_neg
}

PB_evaluate_food_modifiers = {
	set_temp_variable = { temp_agriculture_fulfillment_pct = PB_agriculture_fulfillment_pct }
	PB_calculate_food_variables = yes
	set_variable = { PB_agriculture_dynmod_10_20_30 = temp_agriculture_dynmod_10_20_30 }
	set_variable = { PB_agriculture_dynmod_10_20_30_neg = temp_agriculture_dynmod_10_20_30_neg }
	set_variable = { PB_agriculture_dynmod_5_10_20 = temp_agriculture_dynmod_5_10_20 }
	set_variable = { PB_agriculture_dynmod_5_10_20_neg = temp_agriculture_dynmod_5_10_20_neg }
	set_variable = { PB_agriculture_dynmod_0_10_30 = temp_agriculture_dynmod_0_10_30 }
	set_variable = { PB_agriculture_dynmod_0_10_30_neg = temp_agriculture_dynmod_0_10_30_neg }
}

# Scoped to the country that has intel on PREV
PB_evaluate_food_modifiers_intel = {

	if = {
		limit = {
			OR = {
				check_variable = { var = PB_agriculture_intel_var_lower@PREV value = 50 compare = less_than_or_equals }
				check_variable = { var = PB_agriculture_intel_var_lower@PREV value = 80 compare = greater_than_or_equals }
			}
		}
		# Lower
		set_temp_variable = { temp_agriculture_fulfillment_pct = PB_agriculture_intel_var_lower@PREV }
		divide_temp_variable = { temp_agriculture_fulfillment_pct = 100 }
		PB_calculate_food_variables = yes
	}

	if = {
		limit = {
			OR = {
				check_variable = { var = PB_agriculture_intel_var_upper@PREV value = 50 compare = less_than_or_equals }
				check_variable = { var = PB_agriculture_intel_var_upper@PREV value = 80 compare = greater_than_or_equals }
			}
		}
		# Upper
		set_temp_variable = { temp_agriculture_fulfillment_pct = PB_agriculture_intel_var_upper@PREV }
		divide_temp_variable = { temp_agriculture_fulfillment_pct = 100 }
		PB_calculate_food_variables = yes
	}
}

PB_calculate_food_variables = {
	## I hate my life
	## - Scars

	if = {
		limit = {
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.8 compare = greater_than_or_equals }
		}
		set_temp_variable = { temp_agriculture_bonus_progress = temp_agriculture_fulfillment_pct }
		subtract_from_temp_variable = { temp_agriculture_bonus_progress = 0.8 }
		divide_temp_variable = { temp_agriculture_bonus_progress = 0.2 }

		set_temp_variable = { temp_modifier_value = 0.05 }
		multiply_temp_variable = { temp_modifier_value = temp_agriculture_bonus_progress }

		# negative and positive are switched
		set_temp_variable = { temp_agriculture_dynmod_10_20_30 = temp_modifier_value }
		set_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = temp_modifier_value }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30 = -1 }

		set_temp_variable = { temp_agriculture_dynmod_5_10_20 = temp_modifier_value }
		set_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = temp_modifier_value }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20 = -1 }

		set_temp_variable = { temp_agriculture_dynmod_0_10_30 = temp_modifier_value }
		set_temp_variable = { temp_agriculture_dynmod_0_10_30_neg = temp_modifier_value }
		multiply_temp_variable = { temp_agriculture_dynmod_0_10_30 = -1 }
	}
	else_if = {
		limit = {
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.5 compare = less_than_or_equals }
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.3 compare = greater_than }
		}
		set_temp_variable = { temp_agriculture_famine_progress_stage_1 = 0.5 }
		subtract_from_temp_variable = { temp_agriculture_famine_progress_stage_1 = temp_agriculture_fulfillment_pct }
		divide_temp_variable = { temp_agriculture_famine_progress_stage_1 = 0.2 }

		set_temp_variable = { temp_agriculture_dynmod_10_20_30 = 0.1 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30 = temp_agriculture_famine_progress_stage_1 }
		set_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = temp_agriculture_dynmod_10_20_30 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_5_10_20 = 0.05 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20 = temp_agriculture_famine_progress_stage_1 }
		set_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = temp_agriculture_dynmod_5_10_20 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_0_10_30 = 0 }
		set_temp_variable = { temp_agriculture_dynmod_0_10_30_neg = 0 }

	}
	else_if = {
		limit = {
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.3 compare = less_than_or_equals }
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.1 compare = greater_than }
		}
		set_temp_variable = { temp_agriculture_famine_progress_stage_2 = 0.3 }
		subtract_from_temp_variable = { temp_agriculture_famine_progress_stage_2 = temp_agriculture_fulfillment_pct }
		divide_temp_variable = { temp_agriculture_famine_progress_stage_2 = 0.2 }

		set_temp_variable = { temp_agriculture_dynmod_10_20_30 = 0.1 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30 = temp_agriculture_famine_progress_stage_2 }
		add_to_temp_variable = { temp_agriculture_dynmod_10_20_30 = 0.1 }
		set_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = temp_agriculture_dynmod_10_20_30 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_5_10_20 = 0.05 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20 = temp_agriculture_famine_progress_stage_2 }
		add_to_temp_variable = { temp_agriculture_dynmod_5_10_20 = 0.05 }
		set_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = temp_agriculture_dynmod_5_10_20 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_0_10_30 = 0.1 }
		multiply_temp_variable = { temp_agriculture_dynmod_0_10_30 = temp_agriculture_famine_progress_stage_2 }
		set_temp_variable = { temp_agriculture_dynmod_0_10_30_neg = temp_agriculture_dynmod_0_10_30 }
		multiply_temp_variable = { temp_agriculture_dynmod_0_10_30_neg = -1 }
	}
	else_if = {
		limit = {
			check_variable = { var = temp_agriculture_fulfillment_pct value = 0.1 compare = less_than_or_equals }
		}
		set_temp_variable = { temp_agriculture_famine_progress_stage_3 = 0.1 }
		subtract_from_temp_variable = { temp_agriculture_famine_progress_stage_3 = temp_agriculture_fulfillment_pct }
		divide_temp_variable = { temp_agriculture_famine_progress_stage_3 = 0.1 }

		set_temp_variable = { temp_agriculture_dynmod_10_20_30 = 0.1 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30 = temp_agriculture_famine_progress_stage_3 }
		add_to_temp_variable = { temp_agriculture_dynmod_10_20_30 = 0.2 }
		set_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = temp_agriculture_dynmod_10_20_30 }
		multiply_temp_variable = { temp_agriculture_dynmod_10_20_30_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_5_10_20 = 0.1 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20 = temp_agriculture_famine_progress_stage_3 }
		add_to_temp_variable = { temp_agriculture_dynmod_5_10_20 = 0.1 }
		set_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = temp_agriculture_dynmod_5_10_20 }
		multiply_temp_variable = { temp_agriculture_dynmod_5_10_20_neg = -1 }

		set_temp_variable = { temp_agriculture_dynmod_0_10_30 = 0.2 }
		multiply_temp_variable = { temp_agriculture_dynmod_0_10_30 = temp_agriculture_famine_progress_stage_3 }
