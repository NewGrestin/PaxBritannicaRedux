DAS_civilwar_ready = { 
	# SSL BUFFS
	if = { 
		limit = {
			check_variable = {
				var = GER.FernostReadiness
				value = 8
				compare = greater_than_or_equals
			}
		}
		SSL = { add_ideas = DAS_Fernost_Buff_MAX }
	}
	else_if = {
		limit = {
			check_variable = {
				var = GER.FernostReadiness
				value = 5
				compare = greater_than_or_equals
			}
		}
		SSL = { add_ideas = DAS_Fernost_Buff_MED }
	}
	else_if = {
		limit = {
			check_variable = {
				var = GER.FernostReadiness
				value = 3
				compare = greater_than_or_equals
			}
		}
		SSL = { add_ideas = DAS_Fernost_Buff_LOW }
	}
	else = {
		SSL = {
			add_stability = -0.3
			add_war_support = -0.2
		}
		SSL = { add_ideas = DAS_Fernost_Buff_LOW }
	}

	# DAS BUFFS
	if = { 
		limit = {
			check_variable = {
				var = DAS.PacificReadiness
				value = 8
				compare = greater_than_or_equals
			}
		}
		DAS = { add_ideas = DAS_Pacific_Buff_MAX }
	}
	else_if = {
		limit = {
			check_variable = {
				var = DAS.PacificReadiness
				value = 5
				compare = greater_than_or_equals
			}
		}
		DAS = { add_ideas = DAS_Pacific_Buff_MED }
	}
	else_if = {
		limit = {
			check_variable = {
				var = DAS.PacificReadiness
				value = 3
				compare = greater_than_or_equals
			}
		}
		DAS = { add_ideas = DAS_Pacific_Buff_LOW }
	}
	else = {
		DAS = {
			add_stability = -0.3
			add_war_support = -0.2
		}
		DAS = { add_ideas = DAS_Pacific_Buff_LOW }
	}
}
