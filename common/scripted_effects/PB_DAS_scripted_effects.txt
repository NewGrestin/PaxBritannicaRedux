DAS_start_civil_war = {
	set_global_flag = DAS_civil_war
	GER = {
		puppet = {
        target = SSL
        end_wars = no
        }
	}
	DAS = {
		set_cosmetic_tag = DAS_FREE
		declare_war_on = {
			target = SSL
			type = civil_war
		}
		set_capital = {
			state = 627 #Mindanao
		}
	}
	every_core_state = {
		add_core_of = SSL
	}
	every_owned_state = {
		limit = { is_claimed_by = DAS }
		add_claim_by = SSL
	}
	SSL = {
	    set_cosmetic_tag = DAS_FERNOST
		set_country_flag = DAS_is_Fernost
		inherit_technology = DAS
		set_capital = {
			state = 647 #Palau
		}
		load_oob = "DAS_Fernost"
	}
}
