DAS_start_civil_war = {
	set_global_flag = DAS_civil_war
	DAS = {
		set_cosmetic_tag = DAS_FREE
		declare_war_on = {
			target = SSL
			type = civil_war
		}
		set_capital = {
			state = 627 #Mindanao
		}
	}
	every_core_state = {
		add_core_of = SSL
	}
	every_owned_state = {
		limit = { is_claimed_by = DAS }
		add_claim_by = SSL
	}
	every_owned_state = {
		limit = { has_state_flag = DAS_Fernost_Owner_SSL }
		transfer_state_to = SSL
		create_unit = {
			division = "name = \"Fernost Commandos\" division_template = \"Infanterie-Division\" start_experience_factor = 0.5 start_equipment_factor = 1"
			owner = SSL
		}
	}
	SSL = {
		set_cosmetic_tag = DAS_FERNOST
		set_country_flag = DAS_is_Fernost
		inherit_technology = DAS
		set_capital = {
			state = 647 #Palau
		}
		load_oob = "DAS_Fernost"
	}
}

DAS_Raise_Bund = {
	if = {
		limit = {
			DAS = {
				has_idea = PB_Idea_DAS_Bundestag_Patience_1
			}
