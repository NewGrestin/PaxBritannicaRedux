### INCREASE/DECREASE LEVEL ###
increase_poverty_rate_level = {
    custom_effect_tooltip = increase_poverty_rate_level_tt
    hidden_effect = { country_event = PB_social_development.20 }
    if = {
        limit = {
            has_idea = PB_socdev_extreme_poverty
        }
        swap_ideas = {
            remove_idea = PB_socdev_extreme_poverty
            add_idea = PB_socdev_rampant_poverty
        }
    }
    else_if = {
        limit = {
            has_idea = PB_socdev_rampant_poverty
        }
        swap_ideas = {
            remove_idea = PB_socdev_rampant_poverty
            add_idea = PB_socdev_widespread_poverty
        }
    }
    else_if = {
        limit = {
            has_idea = PB_socdev_widespread_poverty
        }
        swap_ideas = {
            remove_idea = PB_socdev_widespread_poverty
            add_idea = PB_socdev_noticeable_poverty
        }
    }
    else_if = {
        limit = {
            has_idea = PB_socdev_noticeable_poverty
        }
        swap_ideas = {
            remove_idea = PB_socdev_noticeable_poverty
            add_idea = PB_socdev_minor_poverty
        }
    }
}
