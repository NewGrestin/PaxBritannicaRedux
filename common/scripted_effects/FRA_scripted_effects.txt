FRA_calculate_weekly_factionalism = {
	clamp_variable = {
	    var = FRA_factionalism_var
	    min = 0
	    max = 100
    }
	if = {
		limit = {
			check_variable = { var = FRA_factionalism_var value = 75 compare = greater_than_or_equals }
			NOT = { has_idea = FRA_high_factionalism }
		}
		hidden_effect = {
			remove_ideas = {
				FRA_low_factionalism
				FRA_medium_fervour
			}
			add_ideas = FRA_medium_factionalism
		}
	}
	else_if = {
		limit = {
			check_variable = { FRA_factionalism_var < 75 }
			check_variable = { FRA_factionalism_var > 50 }
			NOT = { has_idea = FRA_medium_factionalism }
		}
		hidden_effect = {
			remove_ideas = {
				FRA_low_factionalism
				FRA_high_factionalism
			}
			add_ideas = FRA_medium_factionalism
		}
	}
	else_if = {
		limit = {
			check_variable = { var = FRA_factionalism_var value = 50 compare = less_than_or_equals }
			NOT = { has_idea = FRA_low_factionalism }
		}
		hidden_effect = {
			remove_ideas = {
				FRA_medium_factionalism
				FRA_high_factionalism
			}
			add_ideas = FRA_low_factionalism
		}
	}
}

FRA_define_doriot_dissonance = {
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left > 1 }
			check_variable = { FRA_doriot_dissonance_left < 3 }
			check_variable = { FRA_doriot_dissonance_right < 2 }
		}
		FRA = { save_event_target_as = political_dissonance_left }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left > 2 }
			check_variable = { FRA_doriot_dissonance_right < 1 }
		}
		FRA = { save_event_target_as = political_dissonance_far_left }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left < 2 }
			check_variable = { FRA_doriot_dissonance_right > 1 }
			check_variable = { FRA_doriot_dissonance_right < 3 }
		}
		FRA = { save_event_target_as = political_dissonance_right }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left < 1 }
			check_variable = { FRA_doriot_dissonance_right > 2 }
		}
		FRA = { save_event_target_as = political_dissonance_far_right }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left < 2 }
			check_variable = { FRA_doriot_dissonance_right < 2 }
			OR = {
				NOT = {
					AND = {
						has_completed_focus = FRA_audace
						has_completed_focus = FRA_la_terre
					}
					AND = {
						has_completed_focus = FRA_plebeians_tribune
						has_completed_focus = FRA_nation_first_son
					}
				}
			}
		}
		FRA = { save_event_target_as = political_dissonance_center }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left > 0 }
			check_variable = { FRA_doriot_dissonance_left < 3 }
			check_variable = { FRA_doriot_dissonance_right > 0 }
			check_variable = { FRA_doriot_dissonance_right < 3 }
			NOT = {
				AND = {
					has_completed_focus = FRA_class_collaboration
					OR = {
						has_completed_focus = FRA_plebeians_tribune
						has_completed_focus = FRA_nation_first_son
					}
				}
			}
		}
		FRA = { save_event_target_as = political_dissonance_corporatist }
	}
	
	if = {
		limit = {
			check_variable = { FRA_doriot_dissonance_left > 2 }
			check_variable = { FRA_doriot_dissonance_right > 2 }
		}
		FRA = { save_event_target_as = political_dissonance_confusion }
	}
}
