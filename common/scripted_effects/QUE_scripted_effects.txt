	custom_effect_tooltip = PB_QUE_increase_tensions_tt

			swap_ideas = {
				remove_idea = QUE_Anglois_Tensions_Very_Low
				add_idea = QUE_Anglois_Tensions_Low
			}
		}
		ELSE_IF = {
			limit = {

			}
			swap_ideas = {

		ELSE_IF = {

			}
		}
		ELSE_IF = {
			limit = {
				has_idea = QUE_Anglois_Tensions_Medium
			}
			swap_ideas = {
				remove_idea = QUE_Anglois_Tensions_Medium

			}
		}
		ELSE_IF = {
