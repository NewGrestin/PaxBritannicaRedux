atrbt_buff_authority = {
    if = {
        limit = { has_country_leader_with_trait = PB_Authority_5 }
        # already at max, do nothing
    }
    else_if = {
        limit = { has_country_leader_with_trait = PB_Authority_4 }
        remove_country_leader_trait = PB_Authority_4
        add_country_leader_trait = PB_Authority_5
    }
    else_if = {
        limit = { has_country_leader_with_trait = PB_Authority_3 }
        remove_country_leader_trait = PB_Authority_3
        add_country_leader_trait = PB_Authority_4
    }
    else_if = {
        limit = { has_country_leader_with_trait = PB_Authority_2 }
        remove_country_leader_trait = PB_Authority_2
        add_country_leader_trait = PB_Authority_3
    }
    else_if = {
        limit = { has_country_leader_with_trait = PB_Authority_1 }
        remove_country_leader_trait = PB_Authority_1
        add_country_leader_trait = PB_Authority_2
    }
    else = {
        add_country_leader_trait = PB_Authority_1
    }
}
