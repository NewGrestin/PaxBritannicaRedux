PB_give_to_indian_unifier = {
	if = {
		limit = {
			ROOT = {
				has_monarchist_government = yes
				OR = {
					has_government = rc
					has_government = ni
				}
			}
			MUG = {
				exists = yes
				is_subject = no
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = MUG
				transfer_state_to = MUG
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				has_monarchist_government = yes
				OR = {
					has_government = rc
					has_government = ni
				}
			}
			MUG = {
				exists = no
			}
			AWD = {
				exists = yes
				OR = {
					is_subject = no
					is_subject_of = NEP
				}
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = AWD
				transfer_state_to = AWD
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					has_government = rc
					has_government = ni
					has_government = ds
				}
				tag = FRA
			}
			BHP = {
				exists = yes
				is_subject = no
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = BHP
				transfer_state_to = BHP
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				has_republican_government = yes
				has_government = jc
			}
			FOM = {
				exists = yes
				is_subject = no
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = FOM
				transfer_state_to = FOM
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				has_republican_government = yes
				has_government = jc
			}
			FOM = {
				exists = no
			}
			MAR = {
				exists = yes
				OR = {
					has_government = jc
					has_government = ss
				}
				is_subject = no
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = MAR
				transfer_state_to = MAR
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					tag = ENG
					is_ally_with = ENG
					is_subject_of = ENG
				}
			}
			ENG = {
				is_subject = no
			}
			RAJ = {
				exists = yes
				is_subject_of = ENG
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = RAJ
				transfer_state_to = RAJ
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					tag = ENG
					is_ally_with = ENG
					is_subject_of = ENG
				}
			}
			ENG = {
				is_subject = no
			}
			RAJ = {
				exists = no
			}
			BAN = {
				exists = yes
				is_subject_of = ENG
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = BAN
				transfer_state_to = BAN
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				OR = {
					tag = ENG
					is_ally_with = ENG
					is_subject_of = ENG
				}
			}
			ENG = {
				is_subject = no
			}
			RAJ = {
				exists = no
			}
			BAN = {
				exists = no
			}
			JBP = {
				exists = yes
				is_ally_with = ENG
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = JBP
				transfer_state_to = JBP
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else_if = {
		limit = {
			ROOT = {
				has_country_flag = PB_Techno_Supremacy
			}
			BDK = {
				exists = yes
				is_subject = no
			}
		}
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = BDK
				transfer_state_to = BDK
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
	else = {
		every_controlled_state = {
			if = {
				limit = {
					is_owned_by = ROOT
					has_state_flag = PB_Indian_Unifier_State
					is_on_continent = india
				}
				add_claim_by = MAR
				transfer_state_to = MAR
				clr_state_flag = PB_Indian_Unifier_State
			}
		}
	}
}
