SWE_revolt_check_path = { 
    if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation > 99 }
            check_variable = { SWE_revolt_resources > 25 } 
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_republic_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
    else_if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation > 99 }
            check_variable = { SWE_revolt_resources < 25 }
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_victory_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
    else_if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation < 5 }
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_victory_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
}

SWE_courtyard_camarilla_totals = {
    # Combine Hedin + Kron political power contributions
    set_temp_variable = { SWE_pp_total_temp = hedin_political_variable }
    add_to_temp_variable = { SWE_pp_total_temp = kron_political_variable }
    set_variable = { SWE_courtyard_total_political_power_gain = SWE_pp_total_temp }

    # Combine Hedin + Kron stability contributions
    set_temp_variable = { SWE_stab_total_temp = hedin_stab_variable }
    add_to_temp_variable = { SWE_stab_total_temp = kron_stability_variable }
    set_variable = { SWE_courtyard_total_stability_factor = SWE_stab_total_temp }
}
SWE_hedin_loyalists_calculations = {   # - ppgain - stability

    if = {
        limit = { check_variable = { SWE_HedinLoyalists_Banned = 1 } }
        set_variable = { hedin_political_variable = 0 }
        set_variable = { hedin_stab_variable = 0 }
    }
    else = {
        # Political Power
        set_temp_variable = {
            hedin_political_variable_temp = {
                value = SWE_HedinLoyalists_Support
                add = -50
                divide = 250
            }
        }
        set_variable = { hedin_political_variable = hedin_political_variable_temp }

        # Stability
        set_temp_variable = {
            hedin_stab_variable_temp = {
                value = SWE_HedinLoyalists_Support
                add = -50
                divide = 250
            }
        }
        set_variable = { hedin_stab_variable = hedin_stab_variable_temp }
    }

    SWE_courtyard_camarilla_totals = yes
}
