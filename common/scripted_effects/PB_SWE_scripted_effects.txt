SWE_revolt_check_path = { 
    if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation > 99 }
            check_variable = { SWE_revolt_resources > 25 } 
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_republic_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
    else_if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation > 99 }
            check_variable = { SWE_revolt_resources < 25 }
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_victory_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
    else_if = { 
        limit = { 
            NOT = { has_country_flag = SWE_revolt_resolved }
            check_variable = { SWE_revolt_escalation < 5 }
        }
        set_country_flag = SWE_revolt_resolved
        SWE_revolt_victory_path = yes
        remove_mission = SWE_debug_trigger_state_mission
    }
}

SWE_hedin_loyalists_calculations = {   # - ppgain - stability

        set_variable = { hedin_stab_variable = 0 }
    }
    else = {

SWE_automation_crisis_calculations = {

	#Political Power Gain
	set_temp_variable = {
		trans_pp_temp = {
			value = SWE_crime_violent_rate
			divide = 100
			multiply = -1.0
		}
	}
	set_variable = { SWE_autocrisis_political_power_gain_variable = trans_pp_temp }

	#Stability
	set_temp_variable = {
		trans_stab_unemp_temp = {
			value = SWE_unemployment_rate
			subtract = 50000
			divide = 5000000   # capped contribution: ~0.20 max at 1.05M unemployment
			multiply = -1
		}
	}
	set_temp_variable = {
		trans_stab_crime_temp = {
			value = SWE_crime_violent_rate
			divide = 100
			multiply = -0.25   # capped contribution: ~-0.25 max at crime_rate = 100
		}
	}
	add_to_temp_variable = { trans_stab_unemp_temp = trans_stab_crime_temp }
	set_variable = { SWE_autocrisis_total_stability_factor_variable = trans_stab_unemp_temp }

	# safety clamp so no combination of edge-case values can exceed -35%
	clamp_variable = {
		var = SWE_autocrisis_total_stability_factor_variable
		min = -0.35
		max = 0.10
	}

	#Consumer Goods
	set_temp_variable = {
		trans_congoods_temp = {
			value = SWE_unemployment_rate
			subtract = 50000
			divide = 3000000
		}
	}
	set_variable = { SWE_autocrisis_consumer_goods_factor_variable = trans_congoods_temp }

	#Food Consumption Factor
	set_temp_variable = {
		trans_food_temp = {
			value = SWE_unemployment_rate
			subtract = 50000
			divide = 4000000
		}
	}
	set_variable = { SWE_autocrisis_food_consumption_variable = trans_food_temp }

	#Poverty Growth (SOCDEV_POV_growth_monthly)
	set_temp_variable = {
        trans_poverty_unemp_temp = {
            value = SWE_unemployment_rate
            subtract = 50000
            divide = 2000000
        }
    }
    multiply_temp_variable = { trans_poverty_unemp_temp = -1 }

    set_temp_variable = {
        trans_poverty_crime_temp = {
            value = SWE_crime_violent_rate
            divide = 100
            multiply = 0.3
        }
    }
    multiply_temp_variable = { trans_poverty_crime_temp = -1 }

    add_to_temp_variable = { trans_poverty_unemp_temp = trans_poverty_crime_temp }
    set_variable = { SWE_autocrisis_poverty_variable = trans_poverty_unemp_temp }
    clamp_variable = {
        var = SWE_autocrisis_poverty_variable
        min = -0.8
        max = 0
    }
}
