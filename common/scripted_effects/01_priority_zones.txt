### Priority Zones
###
### Each country has a list of states/regions which it will take in a peace conference even if it
### doesn't have claims/cores. Other countries in their faction will let them take these areas unless there's a conflict.
### If they're a minor, or they're in a faction but not the leader, they won't expand outside of this ambition area unless
### there's no-one involved in the peace conference who can take the land.
###
### If a country is NOT in this list, then it only takes cores/claims.
###

create_priority_zones = {
	# FRA - France
	if = {
		limit = {
			OR = {
				is_core_of = FRA
				is_claimed_by = FRA
				state = 1025
				state = 50
			}
		}
		set_variable = { priority_zone_@FRA = 1 }
	}
	# GER - Germany
	if = {
		limit = {
			OR = {
				is_core_of = GER
				is_claimed_by = GER
				region = 22
				is_prussia = yes
			}
		}
		set_variable = { priority_zone_@GER = 1 }
	}
	# SPR - Spain
	if = {
		limit = {
			OR = {
				is_core_of = SPR
				is_claimed_by = SPR
				is_morocco = yes
			}
		}
		set_variable = { priority_zone_@SPR = 1 }
	}
	# BAL - Baltic
	if = {
		limit = {
			OR = {
				is_core_of = BAL
				is_claimed_by = BAL
				region = 37
			}
		}
		set_variable = { priority_zone_@BAL = 1 }
	}
	# UKR - Ukraine
	if = {
		limit = {
			OR = {
				is_core_of = UKR
				is_claimed_by = UKR
				region = 130
				state = 225
				state = 220
				state = 240
				state = 260
				state = 324
				state = 1390
				state = 1387
				state = 218
			}
		}
		set_variable = { priority_zone_@BAL = 1 }
	}
}
