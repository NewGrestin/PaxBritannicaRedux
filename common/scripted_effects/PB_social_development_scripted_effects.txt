set_starting_social_development_variables = {

    # Monthly SOCDEV base change value, generic is 0
    set_variable = { SOCDEV_POV_monthly_change = 0 } 
    set_variable = { SOCDEV_IND_monthly_change = 0 } 
    set_variable = { SOCDEV_LIT_monthly_change = 0 } 
    set_variable = { SOCDEV_ECON_monthly_change = 0 }
    set_variable = { SOCDEV_MIL_monthly_change = 0 } 
    set_variable = { SOCDEV_WEST_monthly_change = 0 } 

    # Initial SOCDEV progress value, generic is 0
    set_variable = { SOCDEV_POV_count = 0 }  
    set_variable = { SOCDEV_IND_count = 0 }
    set_variable = { SOCDEV_LIT_count = 0 }
    set_variable = { SOCDEV_ECON_count = 0 }
    set_variable = { SOCDEV_MIL_count = 0 }
    set_variable = { SOCDEV_WEST_count = 0 }

    # Real SOCDEV change (base + modifiers), generic is 0
    set_variable = { SOCDEV_POV_real_change = 0 }  
    set_variable = { SOCDEV_IND_real_change = 0 }
    set_variable = { SOCDEV_LIT_real_change = 0 }
    set_variable = { SOCDEV_ECON_real_change = 0 }
    set_variable = { SOCDEV_MIL_real_change = 0 }
    set_variable = { SOCDEV_WEST_real_change = 0 }

    # Effectiveness SOCDEV variable multiplier, base is 100%
    set_variable = { SOCDEV_POV_eff_var = 1 }  
    set_variable = { SOCDEV_IND_eff_var = 1 }
    set_variable = { SOCDEV_LIT_eff_var = 1 }
    set_variable = { SOCDEV_ECON_eff_var = 1 }
    set_variable = { SOCDEV_MIL_eff_var = 1 }
    set_variable = { SOCDEV_WEST_eff_var = 1 }

    # Campaigns #
    set_variable = { socdev_active_campaigns = 0 }
    set_variable = { socdev_active_campaigns_cap = 1 }

    socdev_initial_campaign_modifiers_set = yes
}

social_development_calculate_efficiency = {
    # Recalibration: reset real_change and effectivness totals before calculation ---
    set_variable = { SOCDEV_POV_real_change = 0 }  
	set_variable = { SOCDEV_IND_real_change = 0 }  
	set_variable = { SOCDEV_LIT_real_change = 0 }  
	set_variable = { SOCDEV_ECON_real_change = 0 } 
	set_variable = { SOCDEV_MIL_real_change = 0 }  
	set_variable = { SOCDEV_WEST_real_change = 0 }  

    set_variable = { SOCDEV_POV_eff_var_total = 0 }
	set_variable = { SOCDEV_IND_eff_var_total = 0 }
	set_variable = { SOCDEV_LIT_eff_var_total = 0 }
	set_variable = { SOCDEV_ECON_eff_var_total = 0 }
	set_variable = { SOCDEV_MIL_eff_var_total = 0 }
	set_variable = { SOCDEV_WEST_eff_var_total = 0 }

    # Real change calculated initially from monthly change
    add_to_variable = { SOCDEV_POV_real_change = SOCDEV_POV_monthly_change }
