    add_to_variable = { SOCDEV_LIT_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_ECON_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_MIL_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_CIV_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }

    if = {
        limit = { check_variable = { SOCDEV_POV_eff_var_total < 0 } }
        set_variable = { SOCDEV_POV_eff_var_total = 0 }

    if = {

        set_variable = { SOCDEV_CIV_eff_var_total = 0 }
