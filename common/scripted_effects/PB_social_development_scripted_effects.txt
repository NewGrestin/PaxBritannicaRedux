    set_variable = { SOCDEV_POV_real_change = 0 }  
	set_variable = { SOCDEV_IND_real_change = 0 }  
	set_variable = { SOCDEV_LIT_real_change = 0 }

    set_variable = { SOCDEV_POV_eff_var_total = SOCDEV_POV_eff_var }
	set_variable = { SOCDEV_IND_eff_var_total = SOCDEV_IND_eff_var }
	set_variable = { SOCDEV_LIT_eff_var_total = SOCDEV_LIT_eff_var }
	set_variable = { SOCDEV_ECON_eff_var_total = SOCDEV_ECON_eff_var }
	set_variable = { SOCDEV_MIL_eff_var_total = SOCDEV_MIL_eff_var }
	set_variable = { SOCDEV_WEST_eff_var_total = SOCDEV_WEST_eff_var }

    add_to_variable = { SOCDEV_POV_eff_var_total = modifier@SOCDEV_POV_growth_efficiency }
	add_to_variable = { SOCDEV_IND_eff_var_total = modifier@SOCDEV_IND_growth_efficiency }

    add_to_variable = { SOCDEV_IND_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_LIT_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_ECON_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_MIL_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }
    add_to_variable = { SOCDEV_WEST_eff_var_total = modifier@SOCDEV_ALL_growth_efficiency }


    # Clamps and recalibrates variable if needed

    if = {
        limit = { check_variable = { SOCDEV_POV_eff_var_total < 0 } }
        set_variable = { SOCDEV_POV_eff_var_total = 0 }
    }
    if = {
        limit = { check_variable = { SOCDEV_IND_eff_var_total < 0 } }
        set_variable = { SOCDEV_IND_eff_var_total = 0 }
    }
    if = {
        limit = { check_variable = { SOCDEV_LIT_eff_var_total < 0 } }
        set_variable = { SOCDEV_LIT_eff_var_total = 0 }

    }
    if = {
        limit = { check_variable = { SOCDEV_WEST_eff_var_total < 0 } }
        set_variable = { SOCDEV_WEST_eff_var_total = 0 }

calculate_SOCDEV_industry = {

    # =====================================================
    # 1. REMOVE LAST VALUE (UNDO PREVIOUS TICK)
    # =====================================================
    subtract_from_variable = {
        SOCDEV_IND_industry_monthly = last_SOCDEV_IND_base_total_contribution
    }

    # =====================================================
    # 2. EFFECTIVE STATES PENALTY (UNCHANGED LOGIC)
    # =====================================================

    set_variable = { SOCDEV_effective_states = num_of_core_states }

    set_variable = { SOCDEV_owned_temp = num_of_owned_controlled_states }
    subtract_from_variable = { SOCDEV_owned_temp = num_of_core_states }
    multiply_variable = { SOCDEV_owned_temp = 0.5 }
    add_to_variable = { SOCDEV_effective_states = SOCDEV_owned_temp }

    set_variable = { SOCDEV_occupied_temp = num_of_controlled_states }
    subtract_from_variable = { SOCDEV_occupied_temp = num_of_owned_controlled_states }
    multiply_variable = { SOCDEV_occupied_temp = 0.25 }
    add_to_variable = { SOCDEV_effective_states = SOCDEV_occupied_temp }

    if = {
        limit = { check_variable = { SOCDEV_effective_states < 1 } }
        set_variable = { SOCDEV_effective_states = 1 }
    }

    set_variable = { SOCDEV_state_penalty = SOCDEV_effective_states }
    multiply_variable = { SOCDEV_state_penalty = 0.04 }
    add_to_variable = { SOCDEV_state_penalty = 0.75 }


    # =====================================================
    # 3. CIVILIAN INDUSTRY (MAIN DRIVER)
    # =====================================================
    set_variable = { SOCDEV_civ = num_of_civilian_factories }

    multiply_variable = { SOCDEV_civ = 0.06 }

    clamp_variable = {
        var = SOCDEV_civ
        min = 0
        max = 6
    }


    # =====================================================
    # 4. MILITARY INDUSTRY (WEAKER)
    # =====================================================
    set_variable = { SOCDEV_mil = num_of_military_factories }

    multiply_variable = { SOCDEV_mil = 0.06 }
    multiply_variable = { SOCDEV_mil = 0.4 }

    clamp_variable = {
        var = SOCDEV_mil
        min = 0
        max = 2.5
    }


    # =====================================================
    # 5. NAVAL INDUSTRY (MEDIUM)
    # =====================================================
    set_variable = { SOCDEV_nav = num_of_naval_factories }

    multiply_variable = { SOCDEV_nav = 0.06 }
    multiply_variable = { SOCDEV_nav = 0.6 }
    clamp_variable = {
        var = SOCDEV_mil
        min = 0
        max = 1.5
    }


    # =====================================================
    # 6. COMBINE TOTAL INDUSTRY IMPACT
    # =====================================================
    set_variable = { SOCDEV_IND_industry_monthly = SOCDEV_civ }
    add_to_variable = { SOCDEV_IND_industry_monthly = SOCDEV_mil }
    add_to_variable = { SOCDEV_IND_industry_monthly = SOCDEV_nav }


    # =====================================================
    # 7. APPLY STATE PENALTY (SAFE DIVISION)
    # =====================================================
    divide_variable = { SOCDEV_IND_industry_monthly = SOCDEV_state_penalty }

    # =====================================================
    # 8. APPLY SOCDEV MULTIPLIERS AND CHECK FOR MAJOR STATUS
    # =====================================================
    if = {
        limit = { has_idea = PB_socdev_subsistence_agrarianism }
        multiply_variable = { SOCDEV_IND_industry_monthly = 1.1 }
    }
    if = {
        limit = { has_idea = PB_socdev_agrarianism }
        multiply_variable = { SOCDEV_IND_industry_monthly = 1 }
    }
    if = {
        limit = { has_idea = PB_socdev_semi_industrialized }
        multiply_variable = { SOCDEV_IND_industry_monthly = 0.9 }
    }
    if = {
