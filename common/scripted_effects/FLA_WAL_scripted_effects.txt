BEL_setup = {
	hidden_effect = {
		BEL_setup_characters = yes
		BEL_setup_states = yes
	}
}

BEL_setup_characters = {
	FLA = { every_character = { set_character_flag = BEL_flemish_character set_nationality = BEL } }
	WAL = { every_character = { set_character_flag = BEL_walloon_character set_nationality = BEL } }
}

BEL_remove_characters = {
	BEL = {
		every_character = {
			if = { limit = { has_character_flag = BEL_flemish_character } set_nationality = FLA }
			if = { limit = { has_character_flag = BEL_walloon_character } set_nationality = WAL }
		}
	}
}

BEL_setup_states = {
	every_state = {
		limit = {
			OR = {
				is_core_of = WAL
				is_core_of = FLA
			}
		}
		add_core_of = BEL
	}
	every_state = {
		limit = {
			OR = {
				is_claimed_by = WAL
				is_claimed_by = FLA
			}
		}
		add_claim_by = BEL
	}
}

BEL_clear_states = {
	every_state = {
		limit = { is_core_of = BEL }
		remove_core_of = BEL
	}
	every_state = {
		limit = { is_claimed_by = BEL }
		remove_claim_by = BEL
	}
}

FLA_setup = {
	hidden_effect = {
		BEL_remove_characters = yes
		BEL_clear_states = yes
	}
}

WAL_setup = {
	hidden_effect = {
		BEL_remove_characters = yes
		BEL_clear_states = yes
	}
}
