ASK_update_communities = {
	if = {
		limit = { is_ai = no }
		add_to_variable = { ASK_CC_dirty = 1 }
	}

	if = {
		limit = {
			check_variable = { ASK_Russian_CC_Status > 1 }
			NOT = { has_country_flag = ASK_CC_russian_established }
		}
		add_to_variable = { ASK_CC_established = 1 }
	}

	if = {
		limit = {
			check_variable = { ASK_Polish_CC_Status > 1 }

ASK_increase_prisoners_strenght = {
	custom_effect_tooltip = ASK_increase_prisoners_strenght_tt
	hidden_effect = {
		if = {
		limit = {
			THIS = {
				has_dynamic_modifier = { modifier = ASK_Prison_State_1 }
			}
		}
		THIS = {
			remove_dynamic_modifier = { modifier = ASK_Prison_State_1 }
			add_dynamic_modifier = { modifier = ASK_Prison_State_2 }	
		}
	}
		else_if = {
			limit = {
				THIS = {
					has_dynamic_modifier = { modifier = ASK_Prison_State_0 }
				}
			}
			THIS = {
				remove_dynamic_modifier = { modifier = ASK_Prison_State_0 }
				add_dynamic_modifier = { modifier = ASK_Prison_State_1 }	
			}
		}
	}
}
