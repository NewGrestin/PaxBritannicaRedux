setup_custom_puppet_MOR = {
	if = { #Spain
		limit = { tag = SPR }
			set_autonomy = {
				target = MOR
				autonomous_state = autonomy_PB_Garrison_State
				freedom_level = 0.50
			}
			MOR = {
				set_politics = {
					ruling_party = ds
				}
				set_party_name = {
					ideology = ds
					long_name = "Capitanía General de Marruecos"
					name = "Capitanía General de Marruecos"
				}
				kill_country_leader = yes
				create_country_leader = {
					name = "Damaso Berenguer"
					desc = ""
					picture = "Portrait_MOR_Damaso_Berenguer.dds"
					expire = "1965.1.1"
					ideology = DS_VARIANT_15
				}
				set_cosmetic_tag = MOR_SPR_Custody
			}
	}
}
