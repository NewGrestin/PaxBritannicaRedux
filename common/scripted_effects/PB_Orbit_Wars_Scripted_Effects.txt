OW_check_global_failchance_scripted_effect = {
	add_to_variable = { Local_Launch_Failure_Chance = OW_Rocket_Val_failchance }
	add_to_variable = { Local_Launch_Failure_Chance = OW_Fuel_Val_failchance }
	add_to_variable = { Local_Launch_Failure_Chance = OW_Component_Val_failchance }
	
	clamp_variable = {
		var = Local_Launch_Failure_Chance
		min = 0
		max = 100
	}
}

PB_OBW_Calculate_Fail_Chance = {
	if = {
		limit = {
			OR = {
				check_variable = { Local_Launch_Failure_Chance = 0 }
			}
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_None
	}
	
	if = {
		limit = {
			check_variable = { Local_Launch_Failure_Chance > 0 }
			check_variable = { Local_Launch_Failure_Chance < 31 }
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Low
	}

	if = {
		limit = {
			check_variable = { Local_Launch_Failure_Chance > 30 }
			check_variable = { Local_Launch_Failure_Chance < 51 }
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Medium
	}

	if = {
		limit = {
			OR = {
				check_variable = { Local_Launch_Failure_Chance > 50 }
			}
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Very_High
	}
}
