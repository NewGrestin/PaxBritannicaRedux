	set_variable = { rocket_propulsion_selected = -1 }
	set_variable = { rocket_engine_type_selected = -1 }
	set_variable = { rocket_engine_type_frame = 1 }

	set_variable = { rocket_combat_module_frame = 1 }

	unlock_rocket_propulsion_0 = yes
	unlock_rocket_utility_module_0 = yes

	set_variable = { rocket_engine_failure_chance = 0 }
	set_variable = { rocket_propulsion_failure_chance = 0 }
	set_variable = { rocket_special_module_failure_chance = 0 }
	set_variable = { total_rocket_failure_chance = 0 }

	set_variable = { rocket_engine_civ_cost = 0 }
    set_variable = { rocket_propulsion_civ_cost = 0 }
    set_variable = { rocket_special_module_civ_cost = 0 }
    set_variable = { OW_mission_civ_cost = 0 }

	set_variable = { OW_research_current = 0 }
	set_variable = { OW_research_monthly = 10 }
	add_to_array = { array = global.conquesting_space value = THIS }
	OW_calculate_monthly_research = yes
}

create_solar_system_array = {
	clear_array = solar_system_array

	for_loop_effect = {
		end = 10
		value = temp
		add_to_array = { solar_system_array = temp }
	}

	set_variable = { OW_solar_system_selected = 2 }
}

OW_solar_system_select_left = {
    subtract_from_variable = { OW_solar_system_selected = 1 }

    if = {
        limit = {
            check_variable = { OW_solar_system_selected < 0 }
        }
        set_variable = { OW_solar_system_selected = 9 }
    }
}

OW_solar_system_select_right = {
    add_to_variable = { OW_solar_system_selected = 1 }

    if = {
        limit = {
            check_variable = { OW_solar_system_selected > 9 }
        }
        set_variable = { OW_solar_system_selected = 0 }
    }
}
