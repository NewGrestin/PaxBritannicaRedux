initialize_rocket_parts = {
	for_loop_effect = {
		end = 9
		value = temp
		add_to_array = { rocket_special_modules_array = temp }
		add_to_array = { rocket_special_modules_unlocked_array = 0 }
	}
	for_loop_effect = {
		end = 5
		value = temp
		add_to_array = { rocket_propulsion_array = temp }
		add_to_array = { rocket_propulsion_unlocked_array = 0 }
	}
	for_loop_effect = {
		end = 5
		value = temp
		add_to_array = { rocket_engine_array = temp }
		add_to_array = { rocket_engine_unlocked_array = 0 }
	}
	set_variable = { rocket_special_module_selected = -1 }
	set_variable = { rocket_propulsion_selected = -1 }
	set_variable = { rocket_engine_selected = -1 }
	set_variable = { rocket_special_module_frame = 1 }
	set_variable = { rocket_propulsion_frame = 1 }
	set_variable = { rocket_engine_frame = 1 }

	unlock_rocket_engine_test = yes
	unlock_rocket_propulsion_solid_fuel = yes
	unlock_rocket_module_research_module = yes
}

PB_OBW_Calculate_Fail_Chance = {
	if = {
		limit = {
			OR = {
				check_variable = { OW_total_failure_chance = 0 }
			}
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_None
	}
	
	if = {
		limit = {
			check_variable = { OW_total_failure_chance > 0 }
			check_variable = { OW_total_failure_chance < 31 }
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Low
	}

	if = {
		limit = {
			check_variable = { OW_total_failure_chance > 30 }
			check_variable = { OW_total_failure_chance < 51 }
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Medium
	}

	if = {
		limit = {
			OR = {
				check_variable = { OW_total_failure_chance > 50 }
			}
			NOT = {
				has_country_flag = OBW_Critical_Failure_Assured
			}
		}
		set_country_flag = OBW_Critical_Failure_Very_High
	}
}
