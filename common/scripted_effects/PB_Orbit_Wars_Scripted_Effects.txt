initialize_orbit_wars = {
	clear_array = rocket_special_modules_array
	clear_array = rocket_propulsion_array
	clear_array = rocket_engine_type_array
	for_loop_effect = {
		end = 9
		value = temp
		add_to_array = { rocket_special_modules_array = temp }
		add_to_array = { rocket_special_modules_unlocked_array = 0 }
	}
	for_loop_effect = {
		end = 5
		value = temp
		add_to_array = { rocket_propulsion_array = temp }
		add_to_array = { rocket_propulsion_unlocked_array = 0 }
	}
	for_loop_effect = {
		end = 5
		value = temp
		add_to_array = { rocket_engine_type_array = temp }
		add_to_array = { rocket_engine_type_unlocked_array = 0 }
	}
	set_variable = { rocket_special_module_selected = -1 }
	set_variable = { rocket_propulsion_selected = -1 }
	set_variable = { rocket_engine_type_selected = -1 }
	set_variable = { rocket_special_module_frame = 1 }
	set_variable = { rocket_propulsion_frame = 1 }
	set_variable = { rocket_engine_type_frame = 1 }

	unlock_rocket_engine_type_0 = yes
	unlock_rocket_propulsion_0 = yes
	unlock_rocket_special_module_0 = yes

	set_variable = { rocket_engine_failure_chance = 0 }
	set_variable = { rocket_propulsion_failure_chance = 0 }
	set_variable = { rocket_special_module_failure_chance = 0 }
	set_variable = { total_rocket_failure_chance = 0 }

	set_variable = { OW_research_current = 0 }
	set_variable = { OW_research_monthly = 10 }
	add_to_array = { array = global.conquesting_space value = THIS }
	OW_calculate_monthly_research = yes
}
OW_update = { add_to_variable = { orbit_wars_dirty = 1 } }
