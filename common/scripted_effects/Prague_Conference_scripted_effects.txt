	#ITALY#
	add_to_array = { array = global.major_vp value = 159 } #Lombardy
	add_to_array = { array = global.major_vp value = 158 } #Piedmont
	add_to_array = { array = global.major_vp value = 2} #Latium
	add_to_array = { array = global.major_vp value = 117} #Campania
	
	#SWEDEN#
	add_to_array = { array = global.major_vp value = 141} #Svealand
	add_to_array = { array = global.major_vp value = 140} #Vastergotaland
	
	#NETHERLANDS#
	add_to_array = { global.major_vp = 7} #Holland
	
	for_each_loop = {
		array = global.major_vp
		if = {
            limit = {
                owns_state = THIS
            }

			add_political_power = 100
		}
	}
}

Prague_Conf_determine_leading_tag = {
	if = {
		limit = {
			tag = FRA
			FRA.diplo_weight > RUS.diplo_weight
		}
		FRA = {
			country_event = {
				id = Prague_Conference.5
				days = 3
			}
		}
	}
	if = {
		limit = {
			tag = RUS
			RUS.diplo_weight > FRA.diplo_weight
		}
		RUS = {
			country_event = {
				id = Prague_Conference.5
				days = 3
			}
		}
	}
	else = {
		fixed_random_seed = no
		random_list = {
			50 = {
				FRA = {
					country_event = {
						id = Prague_Conference.5
						days = 3
					}
				}
			}
			50 = {
				RUS = {
					country_event = {
						id = Prague_Conference.5
						days = 3
					}
				}
			}
		}
	}
}
