setup_custom_puppet_AME_United = {
	if = { #Britain
		limit = { tag = ENG }
		if = { 
			limit = { 
				has_character = ENG_hrm_victoria
			}
			country_event = { id = britain_victoria.1 days = 9 }
		}
		set_autonomy = {
			target = AME
			autonomous_state = autonomy_constituent_dominion
			freedom_level = 0.50
		}
		AME = {
			load_focus_tree = generic_focus
			set_politics = {
				ruling_party = ds
				elections_allowed = no
				last_election = "1935.12.17"
				election_frequency = 48
			}
			kill_country_leader = yes
			create_country_leader = {
				name = "Claude Auchinleck"
				desc = "PB_Claude_Auchinleck_DESC"
				picture = "Portrait_AME_Claude_Auchinleck.dds"
				expire = "1965.1.1"
				ideology = DS_VARIANT_01
			}
			add_popularity = {
				ideology = ds
				popularity = 0.3
			}
			set_cosmetic_tag = AME_Thirt_Col
		}
	}
}

				}
				set_party_name = {
