setup_custom_puppet_AME = {
	if = { #Spain
		limit = { tag = ENG }
			set_autonomy = {
				target = AME
				autonomous_state = autonomy_constituent_dominion
				freedom_level = 0.50
			}
			AME = {
				set_politics = {
					ruling_party = ds
				}
				set_party_name = {
					ideology = ds
					long_name = "British Administration of the Middle Colonies"
					name = "BA"
				}
				kill_country_leader = yes
				create_country_leader = {
					name = "Bingham Von Der Gelt"
					desc = ""
					picture = GFX_Portrait_SPR_regency_civilian_large
					expire = "1965.1.1"
					ideology = DS_VARIANT_15
				}
				set_cosmetic_tag = AME_ENG_COLONY
			}
	}
}
